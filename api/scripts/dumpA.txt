function bulkIssueItems(payload) {

  const ss = SpreadsheetApp.getActiveSpreadsheet(); 
  const mov = ss.getSheetByName(MOVEMENTS_SHEET); 

  const ts = new Date(); 
  const movementId = generateMovementId(); 

  const itemCondition = payload.condition || "Consumable";

  // Split cart: positive qty = normal issue, negative qty = exchange /
  // unused return (converted to proper RETURN rows so stock, status and
  // active assignments stay consistent).
  const allItems = payload.items || [];
  const issueItems = allItems.filter(item => Number(item.qty) > 0);
  const returnItems = allItems.filter(item => Number(item.qty) < 0);

  // ---------- NORMAL ISSUE LINES ----------
  issueItems.forEach(item => {

    mov.appendRow([
      movementId, 
      ts, 
      item.geo, 
      "ISSUE",  
      item.qty, 
      payload.person,  
      payload.sector || "",  
      payload.maquina || "", 
      payload.generador || "",  
      payload.vehiculo || "", 
      itemCondition,        
      payload.comments || "",  
      payload.userSite || "" 
    ]);

    updateStatus(
      item.geo, 
      "IN_USE", 
      payload.person, 
      payload.sector, 
      ts,
      itemCondition 
    );

  });

  // ---------- NEGATIVE LINES: ADD BACK TO STOCK ----------
  // Logged as RETURN rows (positive qty) for the same person. Stock is
  // credited to the user's site; item status returns to IN_STORE. No
  // HSE signature is required for returned items.
  returnItems.forEach(item => {

    const returnQty = Math.abs(Number(item.qty)) || 0;
    if (returnQty <= 0) return;

    mov.appendRow([
      movementId,
      ts,
      item.geo,
      "RETURN",
      returnQty,
      payload.person,
      payload.sector || "",
      payload.maquina || "",
      payload.generador || "",
      payload.vehiculo || "",
      "Good",
      "[Exchange/Return via Issue] " + (payload.comments || ""),
      payload.userSite || ""
    ]);

    updateStatus(
      item.geo,
      "IN_STORE",
      "",
      "STORE",
      ts,
      "Good"
    );

  });

  console.log("Bulk issue : ", payload.isHSE);

if (payload.isHSE && payload.hseItems && payload.hseItems.length > 0) {
    console.log("Starting HSE signature queueing for movement: " + movementId);

    try {
      const hseSheet = getOrCreateHSESignaturesSheet();

      const person = getPersonById(payload.personId);
      const workerRut = person ? person.rut : "N/A";
      const workerName = person ? person.name : (payload.person || "UNKNOWN");

      // QUEUE the HSE signature for background PDF generation. The PDF is
      // generated/saved to Drive asynchronously via processPendingHSEPDFs()
      // so the user can continue issuing without waiting on the PDF.
      hseSheet.appendRow([
        movementId,                                   // MovementID
        workerRut,                                    // WorkerID
        workerName,                                   // WorkerName
        payload.fingerprintBase64 || "",              // FingerprintBase64
        payload.matchScore || 0,                      // MatchScore
        ts,                                           // MatchTimestamp
        payload.hseItems.length,                      // ItemsCount
        "",                                           // PDFDriveLink (filled when background processing completes)
        "PENDING",                                    // Status
        JSON.stringify(payload.hseItems)              // HSEItemsJSON (for PDF reconstruction)
      ]);

      console.log("HSE signature queued for RUT: " + workerRut + " (movement " + movementId + ")");
    } catch (e) {
      console.error("HSE Queue Error: " + e.toString());
    }
  }

  // Stock: deduct issued items from the user's site
  if (issueItems.length > 0) {
    processInventoryMovement({
      ...payload,
      items: issueItems,
      movementId: movementId,
      userName: payload.person || "System"   
    }, "ISSUE");
  }

  // Stock: credit returned (negative) items back to the user's site
  if (returnItems.length > 0) {
    const absReturnItems = returnItems.map(item => ({
      ...item,
      qty: Math.abs(Number(item.qty)) || 0
    })).filter(item => item.qty > 0);

    if (absReturnItems.length > 0) {
      processInventoryMovement({
        ...payload,
        items: absReturnItems,
        movementId: movementId,
        userName: payload.person || "System"
      }, "RETURN");
    }
  }

  return {
    success: true,
    message: issueItems.length > 0
      ? ("Bulk issue completed" + (returnItems.length > 0 ? " (" + returnItems.length + " item(s) added back)" : ""))
      : "Return via issue completed (" + returnItems.length + " item(s) added back)"
  };
}

function getOrCreateHSESignaturesSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(HSE_SIGN_SHEET);

  if (!sheet) {
    sheet = ss.insertSheet(HSE_SIGN_SHEET);
    sheet.appendRow([
      "MovementID",
      "WorkerID",
      "WorkerName",
      "FingerprintBase64",
      "MatchScore",
      "MatchTimestamp",
      "ItemsCount",
      "PDFDriveLink",
      "Status",
      "HSEItemsJSON"
    ]);
  } else {
    // Ensure the required columns exist (backwards compatibility with older sheets)
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const ensureHeader = (header, insertAt) => {
      if (headers.indexOf(header) === -1) {
        sheet.getRange(1, insertAt).setValue(header);
      }
    };
    ensureHeader("PDFDriveLink", 8);
    ensureHeader("Status", 9);
    ensureHeader("HSEItemsJSON", 10);
  }

  return sheet;
}

 * BACKGROUND PROCESSOR: Reads all PENDING HSE rows, generates the PDF,

 *
 * This is meant to be called fire-and-forget from the frontend after the
 * issuing completes, so the user never has to wait for PDF generation.

function processPendingHSEPDFs() {
  const hseSheet = getOrCreateHSESignaturesSheet();
  const data = hseSheet.getDataRange().getValues();

  if (data.length < 2) {
    console.log("No HSE signature rows to process.");
    return { success: true, processed: 0 };
  }

  const headers = data[0];
  const colMap = {};
  headers.forEach((h, i) => { colMap[String(h).trim()] = i; });

  const idxMovement = colMap["MovementID"];
  const idxWorkerId = colMap["WorkerID"];
  const idxWorkerName = colMap["WorkerName"];
  const idxFingerprint = colMap["FingerprintBase64"];
  const idxScore = colMap["MatchScore"];
  const idxTimestamp = colMap["MatchTimestamp"];
  const idxItemsCount = colMap["ItemsCount"];
  const idxPdfLink = colMap["PDFDriveLink"];
  const idxStatus = colMap["Status"] !== undefined ? colMap["Status"] : 8;
  const idxItemsJson = colMap["HSEItemsJSON"] !== undefined ? colMap["HSEItemsJSON"] : 9;

  let processed = 0;

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const status = String(row[idxStatus] || "").trim().toUpperCase();

    if (status !== "PENDING") continue;

    const movementId = String(row[idxMovement] || "");
    const workerRut = String(row[idxWorkerId] || "");
    const workerName = String(row[idxWorkerName] || "");
    const fingerprintBase64 = String(row[idxFingerprint] || "");
    const matchScore = Number(row[idxScore]) || 0;
    const timestamp = row[idxTimestamp] instanceof Date ? row[idxTimestamp] : new Date(row[idxTimestamp]);

    let hseItems = [];
    try {
      const rawItems = String(row[idxItemsJson] || "").trim();
      if (rawItems) {
        hseItems = JSON.parse(rawItems);
      }
    } catch (e) {
      console.error("Failed to parse HSEItemsJSON for movement " + movementId + ": " + e.toString());
      hseItems = [];
    }

    try {
      // Mark as processing to avoid duplicate attempts on concurrent runs
      hseSheet.getRange(i + 1, idxStatus + 1).setValue("PROCESSING");

      const pdfData = {
        movementId: movementId,
        timestamp: timestamp,
        workerName: workerName,
        workerRut: workerRut,
        fingerprintBase64: fingerprintBase64,
        matchScore: matchScore,
        hseItems: hseItems
      };

      const pdfBlob = generateHSEReceiptPDF(pdfData);
      const driveLink = saveHSEPDFToDrive(movementId, workerRut, workerName, pdfBlob);

      // Update the row with the Drive link and mark as DONE
      hseSheet.getRange(i + 1, idxPdfLink + 1).setValue(driveLink);
      hseSheet.getRange(i + 1, idxStatus + 1).setValue("DONE");

      processed++;
      console.log("HSE PDF generated in background for movement: " + movementId);
    } catch (e) {
      console.error("HSE PDF background processing failed for movement " + movementId + ": " + e.toString());
      // Reset to PENDING so it can be retried on the next run
      hseSheet.getRange(i + 1, idxStatus + 1).setValue("PENDING");
    }
  }

function getBundleItems(bundleId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const bundleSheet = ss.getSheetByName("Bundle_Items");
  if (!bundleSheet) {
    throw new Error("Bundle_Items sheet not found");
  }

  const data = bundleSheet.getDataRange().getValues();
  const headers = data[0];

  const colBundle = headers.indexOf("Bundle ID");
  const colGeo = headers.indexOf("Geo Code");
  const colQty = headers.indexOf("Qty");

  if (colBundle === -1 || colGeo === -1 || colQty === -1) {
    throw new Error("Required columns missing in Bundle_Items");
  }

  const rows = data.slice(1).filter(r => {
    return String(r[colBundle]).trim() === String(bundleId).trim();
  });

  const inventory = getInventoryData();

  const result = rows.map(row => {
    const geo = String(row[colGeo] || "").trim();
    const qty = Number(row[colQty] || 1);

    const inv = inventory.find(x => {
      return String(x.geo || "").trim() === geo;
    }) || {};

    return {
      geo: geo,
      qty: qty,
      es: inv.es || "",
      en: inv.en || "",
      supplier: inv.supplier || "",
      category: inv.category || "General",
      subCategory: inv.subCategory || "",
      pn: inv.p || inv.pn || "",
      fromBundle: true,
      bundleId: bundleId
    };
  });

  return result;
}



/* ============================================================
 * REQUIREMENT NUMBER SEQUENCE (Req No.)
 * ------------------------------------------------------------
 * The Req number comes from the Config sheet ("LastReq" label in Col A,
 * the value in Col B) and is the LAST NUMBER ISSUED, so the next
 * requirement gets that value + 1.
 *
 * It is deliberately NOT derived from the Ordenes row layout: Ordenes
 * stores ONE ROW PER LINE ITEM, so a row-based number advanced by the
 * item count of every requirement (a 124-item requirement jumped the
 * sequence by 124: 1226 -> 1350). Row position is also shifted by draft
 * rows and by deleteDraftRows, which could move the sequence backwards
 * and re-issue an existing number.
 *
 * Rules enforced here:
 *  - one number per REQUIREMENT (per shipment), never per item row;
 *  - the counter is reserved and persisted BEFORE the slow PDF/row/email
 *    work, because processMassOrders has no rollback (an email failure
 *    re-throws after earlier shipments are already written). An aborted
 *    batch may burn a number - a harmless gap - but can never hand the
 *    same number to two requirements. A duplicate Req ID is worse than a
 *    gap: orchestrateRequirementPDF trashes any existing <OrderId>.pdf,
 *    so it would also destroy the other requirement's PDF and cross-link
 *    both requirements in the Quotes/RFQ screens;
 *  - the read-modify-write runs under the script lock, so two
 *    simultaneous submissions cannot both read the same value;
 *  - the counter is still checked against Ordenes Col A, so a counter
 *    that was blanked, reset or hand-edited can only move forward and
 *    skips numbers that are already in use (self-healing, no manual
 *    repair required on the hot path).
 * ============================================================ */

/**
 * Locates (or recreates) the Config cell holding the last issued Req number.
 * Addressed by its Col-A label instead of a hard-coded "B26" so inserting or
 * moving a row in the Config sheet can never silently break requirement
 * creation the way the hard-coded B15/B16 flag reads can.
 * @param {Sheet} configSheet the Config sheet
 * @return {Range} the counter cell (Col B of the "LastReq" row)
 */
function getLastReqCell_(configSheet) {
  if (!configSheet) {
    throw new Error("Config sheet not found - cannot issue the Req number.");
  }

  const lastRow = Math.max(configSheet.getLastRow(), 1);
  const colA = configSheet.getRange(1, 1, lastRow, 1).getValues();

  for (let i = 0; i < colA.length; i++) {
    if (String(colA[i][0] || "").trim().toLowerCase() === "lastreq") {
      return configSheet.getRange(i + 1, 2);
    }
  }

  // Self-healing: the label was deleted or renamed. Recreate it at the end of
  // the table (value is repaired by reserveReqNumbers_) instead of failing the
  // whole requirement submission.
  const newRow = configSheet.getLastRow() + 1;
  console.warn("Config: 'LastReq' label not found - recreated at row " + newRow);
  configSheet.getRange(newRow, 1).setValue("LastReq");
  return configSheet.getRange(newRow, 2);
}

/**
 * Reads Ordenes column A ONCE and reports which Req numbers are already in use.
 * Understands "REQ-0002-SITE-CAT", the legacy plain-number ids written by the
 * old saveRequirementToSheet path, and ignores "DRAFT-" rows (drafts carry a
 * timestamp id, never a Req number).
 * @param {Sheet} sheet the Ordenes sheet
 * @return {{used: Object, max: number, rows: number}} used numbers as a lookup,
 *         the highest number found and how many id rows were scanned
 */
function scanUsedReqNumbers_(sheet) {
  const used = {};
  let max = 0;
  let rows = 0;

  if (!sheet) return { used: used, max: max, rows: rows };

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return { used: used, max: max, rows: rows };

  // Single-column read (not getDataRange): cheap enough to run per submission.
  const colA = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  for (let i = 0; i < colA.length; i++) {
    const raw = String(colA[i][0] || "").trim();
    if (!raw || raw.toUpperCase().indexOf("DRAFT-") === 0) continue;
    rows++;

    let num = NaN;
    const reqMatch = raw.match(/^REQ-0*(\d+)/i);
    if (reqMatch) {
      num = Number(reqMatch[1]);
    } else if (/^\d+$/.test(raw)) {
      num = Number(raw); // legacy numeric Order ids
    }

    if (isFinite(num) && num > 0) {
      used[num] = true;
      if (num > max) max = num;
    }
  }

  return { used: used, max: max, rows: rows };
}
