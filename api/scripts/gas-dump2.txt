===== bgEnqueueJob (BackgroundJobsGS.js) =====
function bgEnqueueJob(type, refId, title, userEmail) {
  var jobId = (type === "QUOTE" ? "Q-" : "REQ-") +
    String(refId || "x").replace(/[^A-Za-z0-9_-]/g, "").slice(0, 24) +
    "-" + Date.now().toString(36) + "-" + Math.floor(Math.random() * 1296).toString(36);
  var job = {
    jobId: jobId,
    type: type,
    refId: String(refId || ""),
    title: String(title || (type + " " + refId)),
    status: "RUNNING",
    progress: 5,
    step: "Sheet saved — finishing in background…",
    createdBy: String(userEmail || ""),
    createdAt: bgNowIso_(),
    updatedAt: bgNowIso_(),
    pdfUrl: "",
    emailSent: null,
    error: ""
  };
  try {
    CacheService.getScriptCache().put(bgJobKey_(jobId), JSON.stringify(job), BG_JOB_TTL_SEC_);
  } catch (e) { /* cache best-effort */ }
  return { jobId: jobId, job: job };
}

===== bgFinishJob (BackgroundJobsGS.js) =====
function bgFinishJob(jobId, success, extra) {
  try {
    var cache = CacheService.getScriptCache();
    var raw = cache.get(bgJobKey_(jobId));
    var job = raw ? JSON.parse(raw) : { jobId: jobId };
    job.status = success ? "DONE" : "ERROR";
    job.progress = success ? 100 : (Number(job.progress) || 50);
    job.updatedAt = bgNowIso_();
    if (extra) {
      if (extra.step) job.step = String(extra.step);
      if (extra.pdfUrl !== undefined) job.pdfUrl = extra.pdfUrl;
      if (extra.emailSent !== undefined) job.emailSent = extra.emailSent;
      if (extra.error !== undefined) job.error = String(extra.error || "");
      if (extra.result !== undefined) job.result = extra.result;
    } else if (!success && !job.step) {
      job.step = "Failed";
    }
    if (success && !extra?.step) job.step = "Done ✓";
    cache.put(bgJobKey_(jobId), JSON.stringify(job), BG_JOB_TTL_SEC_);
    return job;
  } catch (e) {
    return { jobId: jobId, status: success ? "DONE" : "ERROR" };
  }
}

===== bgGetJobs (BackgroundJobsGS.js) =====
function bgGetJobs(jobIds) {
  var out = [];
  try {
    var cache = CacheService.getScriptCache();
    (jobIds || []).forEach(function (id) {
      try {
        var raw = cache.get(bgJobKey_(id));
        if (raw) out.push(JSON.parse(raw));
        else out.push({ jobId: id, status: "UNKNOWN", progress: 0, step: "Expired or unknown job" });
      } catch (e) {
        out.push({ jobId: id, status: "UNKNOWN", progress: 0, step: "Read error" });
      }
    });
  } catch (e) {
    (jobIds || []).forEach(function (id) {
      out.push({ jobId: id, status: "UNKNOWN", progress: 0, step: "Cache unavailable" });
    });
  }
  return out;
}

function bulkSavePeople(payloadArray, mode) {
  if (!payloadArray || !Array.isArray(payloadArray) || payloadArray.length === 0) {
    return { success: false, message: "No person entries found." };
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(PEOPLE_SHEET);
  
  const data = sheet.getDataRange().getValues();
  const rutMap = {};
  
  if (mode === "upsert" && data.length > 1) {
    for (let i = 1; i < data.length; i++) {
      const existingRut = String(data[i][6]).trim(); // Column G is index 6
      if (existingRut) {
        rutMap[existingRut] = i + 1;
      }
    }
  }

  let baseTimestamp = new Date().getTime();
  const rowsToAppend = [];
  let updateCount = 0;

  payloadArray.forEach((person, index) => {
    let isActive = true;
    if (person.active) {
      const statusStr = person.active.trim().toLowerCase();
      if (statusStr === "false" || statusStr === "inactive") isActive = false;
    }

    const targetRow = person.rut ? rutMap[person.rut] : null;

    if (mode === "upsert" && targetRow) {
      sheet.getRange(targetRow, 2, 1, 8).setValues([[
        person.name || "",
        person.department || "",
        isActive,
        person.sector || "", // Column E: Default Sector
        person.notes || "",  // Column F: Notes
        person.rut || "",    // Column G: RUT
        person.biometricTemplate || "", // Column H: BiometricTemplate
        person.email || ""   // Column I: Email
      ]]);
      updateCount++;
    } else {
      const customId = "RD-" + (baseTimestamp + index);
      rowsToAppend.push([
        customId,
        person.name || "",
        person.department || "",
        isActive,
        person.sector || "", // Column E: Default Sector
        person.notes || "",  // Column F: Notes
        person.rut || "",    // Column G: RUT
        person.biometricTemplate || "", // Column H: BiometricTemplate
        person.email || ""   // Column I: Email
      ]);
    }
  });

  if (rowsToAppend.length > 0) {
    const startRow = sheet.getLastRow() + 1;
    sheet.getRange(startRow, 1, rowsToAppend.length, 9).setValues(rowsToAppend);
  }

  let summaryMessage = "";
  if (rowsToAppend.length > 0) summaryMessage += `${rowsToAppend.length} new profiles added. `;
  if (updateCount > 0) summaryMessage += `${updateCount} profiles updated successfully.`;

  return { success: true, message: summaryMessage || "Done." };
}

function deleteDraftRows(draftId) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Ordenes");

    if (!sheet) return { success: false, message: "Ordenes sheet not found" };

    const data = sheet.getDataRange().getValues();
    const rowsToDelete = [];

    for (let i = 1; i < data.length; i++) {
      const orderId = String(data[i][0] || "").trim();
      if (orderId === draftId) {
        rowsToDelete.push(i + 1);
      }
    }

    if (rowsToDelete.length === 0) {
      return { success: false, message: "Draft not found" };
    }

    for (let i = rowsToDelete.length - 1; i >= 0; i--) {
      sheet.deleteRow(rowsToDelete[i]);
    }

    console.log("Deleted " + rowsToDelete.length + " rows for draft: " + draftId);
    return { success: true, deletedCount: rowsToDelete.length };
  } catch (e) {
    console.error("Error deleting draft rows: " + e.toString());
    return { success: false, message: e.toString() };
  }
}

function deleteMinMaxItems(rows) {
  if (!rows || !Array.isArray(rows)) {
    throw new Error("deleteMinMaxItems: rows array required");
  }
  const sheet = getMinMaxSheet_();
  const data = sheet.getDataRange().getValues();
  const toDelete = [];

  rows.forEach((entry) => {
    const site = String(entry.site || "").trim().toUpperCase();
    const geo = String(entry.geo || "").trim().toUpperCase();
    for (let i = 1; i < data.length; i++) {
      if (
        String(data[i][0] || "").trim().toUpperCase() === site &&
        String(data[i][1] || "").trim().toUpperCase() === geo
      ) {
        toDelete.push(i + 1);
        break;
      }
    }
  });

  // Delete from bottom up.
  let deleted = 0;
  toDelete
    .sort((a, b) => b - a)
    .forEach((row) => {
      sheet.deleteRow(row);
      deleted++;
    });

  return { success: true, deleted };
}

function extractInvoicePDFText(payload) {

  try {

    const blob = Utilities.newBlob(
      Utilities.base64Decode(payload.base64),
      "application/pdf",
      payload.fileName
    );

    // TEMP PDF FILE
    const pdfFile =
      DriveApp.createFile(blob);

    // OCR CONVERT TO GOOGLE DOC
    const resource = {
      title: payload.fileName,
      mimeType: MimeType.GOOGLE_DOCS
    };

    const converted =
      Drive.Files.copy(
        resource,
        pdfFile.getId(),
        {
          ocr: true
        }
      );

    // READ OCR TEXT
    const doc =
      DocumentApp.openById(converted.id);

    const text =
      doc.getBody().getText();

    // CLEANUP
    DriveApp.getFileById(pdfFile.getId()).setTrashed(true);
    DriveApp.getFileById(converted.id).setTrashed(true);

    return {
      success: true,
      text: text
    };

  } catch(err) {

    return {
      success: false,
      message: err.toString()
    };

  }

}

===== getAllActiveAssignments (InvMovGS.js) =====
function getAllActiveAssignments() {
  const ss = SpreadsheetApp.getActiveSpreadsheet(); 
  const mov = ss.getSheetByName(MOVEMENTS_SHEET); 

  if (!mov) return []; 
  const data = mov.getDataRange().getValues(); 
  if (data.length < 2) return []; 
  const map = {}; 

  data.slice(1).forEach(r => {
    const movementId = r[0] || ""; 
    const timestamp  = r[1] ? new Date(r[1]).toLocaleString() : ""; 
    const geo        = String(r[2] || "").trim(); 
    const action     = String(r[3] || "").trim().toUpperCase(); 
    const qty        = parseFloat(r[4]) || 0; 
    const person     = String(r[5] || "").trim().toUpperCase(); 
    const condition  = String(r[10] || "").trim().toUpperCase(); 

    if (action === "ISSUE" && condition === "CONSUMABLE") {
      return;
    }

    if (!geo || !person) return; 

    const key = `${person}__${geo}`; 

    if (!map[key]) {
      map[key] = {
        geo, 
        person, 
        issued: 0, 
        returned: 0, 
        active: 0, 
        sector: "", 
        maquina: "", 
        generador: "", 
        vehiculo: "", 
        movementId: "", 
        timestamp: "" 
      };
    }

    map[key].movementId = movementId; 
    map[key].timestamp = timestamp; 
    map[key].sector = String(r[6] || "").trim(); 
    map[key].maquina = String(r[7] || "").trim(); 
    map[key].generador = String(r[8] || "").trim(); 
    map[key].vehiculo = String(r[9] || "").trim(); 

    if (action === "ISSUE") {
      map[key].issued += qty; 
    } else if (action === "RETURN") {
      map[key].returned += qty; 
    }
    map[key].active = map[key].issued - map[key].returned; 
  });

  const results = Object.values(map).filter(x => x.active > 0); 
  console.log("Found active assignments:", results.length); 
  return results; 
}

function getAllDailyMovements() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(MOVEMENTS_SHEET);

    if (!sheet) {
      throw new Error(
        'Sheet "' + MOVEMENTS_SHEET + '" was not found.'
      );
    }

    const data = sheet.getDataRange().getValues();

    console.log("getAllDailyMovements()");
    console.log("Spreadsheet: " + ss.getName());
    console.log("Sheet: " + sheet.getName());
    console.log("Rows found: " + data.length);

    if (data.length <= 1) {
      console.log("No movement rows found.");
      return [];
    }

    const timezone = Session.getScriptTimeZone();
    const results = [];

    // ----- Build inventory lookup: geo -> {en, es} descriptions -----
    const invMap = {};
    try {
      const masterSheet = ss.getSheetByName("Master");
      if (masterSheet) {
        const masterData = masterSheet.getDataRange().getValues();
        // Master columns: 0=geo, 1=pn, 2=category, 3=subcat, 4=brand, 5=en/desc, 6=es/desc
        for (let i = 1; i < masterData.length; i++) {
          const geo = String(masterData[i][0] || "").trim();
          if (geo) {
            invMap[geo] = {
              en: String(masterData[i][5] || ""),
              es: String(masterData[i][6] || "")
            };
          }
        }
      }
    } catch (e) {
      console.error("Error loading inventory descriptions: " + e.toString());
    }

    // ----- Build HSE PDF link lookup: movementId -> PDFDriveLink -----
    const hseMap = {};
    try {
      const hseSheet = ss.getSheetByName(HSE_SIGN_SHEET);
      if (hseSheet) {
        const hseData = hseSheet.getDataRange().getValues();
        if (hseData.length > 1) {
          const headers = hseData[0];
          const colMap = {};
          headers.forEach(function(h, i) { colMap[String(h).trim()] = i; });
          const idxMovement = colMap["MovementID"];
          const idxPdfLink = colMap["PDFDriveLink"];
          const idxStatus = colMap["Status"];
          for (let i = 1; i < hseData.length; i++) {
            const row = hseData[i];
            const movementId = String(row[idxMovement] || "").trim();
            const status = String(row[idxStatus] || "").trim().toUpperCase();
            const pdfLink = String(row[idxPdfLink] || "").trim();
            if (movementId && status === "DONE" && pdfLink) {
              hseMap[movementId] = pdfLink;
            }
          }
        }
      }
    } catch (e) {
      console.error("Error loading HSE PDF links: " + e.toString());
    }

    // Skip header
    for (let i = 1; i < data.length; i++) {

      const row = data[i];

      const movementId = String(row[0] || "").trim();
      const rawTimestamp = row[1];

      // Ignore completely empty rows
      if (!movementId || !rawTimestamp) {
        continue;
      }

      const rowDate = new Date(rawTimestamp);

      if (isNaN(rowDate.getTime())) {
        console.log(
          "Invalid timestamp at row " + (i + 1) + ": " + rawTimestamp
        );
        continue;
      }

      // IMPORTANT:
      // Convert Date to STRING before sending through google.script.run
      const dateStr = Utilities.formatDate(
        rowDate,
        timezone,
        "yyyy-MM-dd"
      );

      const timeStr = Utilities.formatDate(
        rowDate,
        timezone,
        "HH:mm:ss"
      );

      // ISO-like timestamp, but generated in the script timezone
      const timestampStr = Utilities.formatDate(
        rowDate,
        timezone,
        "yyyy-MM-dd'T'HH:mm:ss"
      );

      const inv = invMap[String(row[2] || "").trim()] || {};

      results.push({
        movementId: movementId,
        timestamp: timestampStr,
        dateStr: dateStr,
        timeStr: timeStr,

        itemGeo: String(row[2] || ""),
        type: String(row[3] || "").toUpperCase(),
        qty: Number(row[4]) || 0,

        person: String(row[5] || ""),
        sector: String(row[6] || ""),
        maquina: String(row[7] || ""),
        generador: String(row[8] || ""),
        vehiculo: String(row[9] || ""),

        condition: String(row[10] || ""),
        comments: String(row[11] || ""),

        // Site: col 13. Old rows are backfilled directly in the sheet (manual one-time pass).
        site: String(row[12] || "").trim(),

        descEn: inv.en || "",
        descEs: inv.es || "",
        pdfLink: hseMap[movementId] || ""
      });
    }

    // Newest first
    results.sort(function(a, b) {
      return b.timestamp.localeCompare(a.timestamp);
    });

    console.log("Returning movements: " + results.length);

    return results;

  } catch (err) {

    console.error(
      "Error in getAllDailyMovements: " + err.toString()
    );

    throw new Error(
      "getAllDailyMovements failed: " + err.message
    );
  }
}

    const val = String(row[0] || '').trim();

      result.push({
        _rowIndex: idx + 2, // Sheet row index
        name: val
      });
    }
  });
  
  return result;
}

function normalizeStage(s) {
  if (!s) return "Created";
  const x = String(s).toLowerCase();

  // twins): the supplier-didn't-quote state must never read as "Quoted".

  if (x.includes("po") || x.includes("orden") || x.includes("ordered") || x.includes("comprad")) return "PO Created";

  if (x.includes("rfq") || x.includes("sent")) return "RFQ Issued";
  if (x.includes("pend")) return "Pending";
  return "Created";
}

function getDrafts() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Ordenes");

    if (!sheet) return [];

    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return [];

    const headers = data[0];
    const colMap = {};
    headers.forEach((h, i) => { colMap[String(h).trim()] = i; });

    const drafts = {};

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const orderId = String(row[0] || "").trim();
      const estado = String(row[colMap["Estado"] !== undefined ? colMap["Estado"] : 7] || "").trim();

      if (orderId.startsWith("DRAFT-") && estado === "Draft") {
        if (!drafts[orderId]) {
          drafts[orderId] = {
            draftId: orderId,
            fecha: row[1] ? new Date(row[1]).toISOString() : "",
            site: row[colMap["Faena"] !== undefined ? colMap["Faena"] : 4] || "",
            user: row[colMap["User"] !== undefined ? colMap["User"] : 8] || "",
            category: row[colMap["Category"] !== undefined ? colMap["Category"] : 12] || "General",
            subArea: row[colMap["SubArea"]] || "",
            emails: row[colMap["Emails"]] || "",
            subAreaNo: row[colMap["SubArea No"]] || "",
            notes: row[colMap["Notes"]] || "",
            items: []
          };
        }

        drafts[orderId].items.push({
          pn: row[colMap["P/N"] !== undefined ? colMap["P/N"] : 2] || "",
          geo: row[colMap["Geo Code"] !== undefined ? colMap["Geo Code"] : 3] || "",
          desc: row[colMap["Desc Es"] !== undefined ? colMap["Desc Es"] : 5] || "",
          es: row[colMap["Desc Es"] !== undefined ? colMap["Desc Es"] : 5] || "",
          qty: row[colMap["Cantidad"] !== undefined ? colMap["Cantidad"] : 6] || 0,
          itemNotes: row[colMap["Item Notes"]] || "",
          supplierNotes: row[colMap["Supplier Notes"]] || "",
          // Attached photos come back as the raw comma-separated cell; the client
          // splits it back into an array of {url,name} on loadDraft().
          images: row[colMap["Item Images"]] || ""
        });
      }
    }

    return Object.values(drafts).sort((a, b) => new Date(b.fecha) - new Date(a.fecha));

===== getEmailSuggestions (Código.js) =====
function getEmailSuggestions(searchTerm) {
  if (!searchTerm || String(searchTerm).trim().length === 0) return [];

  const term = String(searchTerm).trim().toLowerCase();
  const results = [];
  const seen = {};

  const addResult = function(displayName, email, source) {
    if (!email || !displayName) return;
    const key = email.toLowerCase();
    if (Object.prototype.hasOwnProperty.call(seen, key)) return;
    seen[key] = true;
    results.push({
      label: displayName + ' <' + email + '>',
      value: email,
      source: source
    });
  };

  // 1. People sheet (company domain emails only)
  try {
    const people = getPeople() || [];
    people.forEach(function(p) {
      if (!p || !p.email) return;
      const email = String(p.email).toLowerCase().trim();
      if (!isCompanyEmail(email)) return;
      const name = String(p.name || '').trim();
      if (!name) return;
      if (name.toLowerCase().indexOf(term) !== -1 || email.indexOf(term) !== -1) {
        addResult(name, email, 'people');
      }
    });
  } catch (e) {
    console.warn('People sheet suggestion lookup failed:', e.toString());
  }

  // 2. Suppliers (from Validation sheet directly)
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const valSheet = ss.getSheetByName('Validation');
    let suppliers = [];
    if (valSheet) {
      const valData = valSheet.getDataRange().getValues();
      for (let i = 1; i < valData.length; i++) {
        if (valData[i][15]) {
          suppliers.push({ name: String(valData[i][15] || '').trim(), email: String(valData[i][16] || '').trim() });
        }
      }
    }
    suppliers.forEach(function(s) {
      if (!s || !s.email) return;
      const email = String(s.email).toLowerCase().trim();
      if (!email) return;
      const name = String(s.name || '').trim();
      if (!name) return;
      if (name.toLowerCase().indexOf(term) !== -1 || email.indexOf(term) !== -1) {
        addResult(name, email, 'supplier');
      }
    });
  } catch (e) {
    console.warn('Supplier suggestion lookup failed:', e.toString());
  }

  // 3. Microsoft Graph directory (company emails only)
  try {
    const graphResults = searchGraphDirectory(searchTerm);
    graphResults.forEach(function(u) {
      addResult(u.displayName || u.email, u.email, 'graph');
    });
  } catch (e) {
    console.warn('Graph suggestion lookup failed:', e.toString());
  }

  // Sort: Graph first (canonical names), then People, then Suppliers
  const sourceOrder = { graph: 1, people: 2, supplier: 3 };
  results.sort(function(a, b) {
    const sa = sourceOrder[a.source] || 99;
    const sb = sourceOrder[b.source] || 99;
    if (sa !== sb) return sa - sb;
    return (a.label || '').localeCompare(b.label || '');
  });

  return results.slice(0, 10);
}

function getInventoryChunk(startRow, chunkSize) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Master");
    const lastRow = sheet.getLastRow();

    if (startRow > lastRow) {
      return { data: "", total: lastRow - 1, nextRow: -1 };
    }

    const remaining = lastRow - startRow + 1;
    const numRows = Math.min(chunkSize, remaining);

    // Fetch columns 1 to 15
    const values = sheet.getRange(startRow, 1, numRows, 15).getValues();

    // Convert to safe JSON strings per line to avoid transmission errors
    const serialized = values
      .map(function (row) {
        return JSON.stringify(row);
      })
      .join("\n");

    return {
      data: serialized,
      total: lastRow - 1,
      nextRow: startRow + numRows > lastRow ? -1 : startRow + numRows,
    };
  } catch (e) {
    console.error("Backend Error: " + e.toString());
    return null;
  }
}

        estado: r.estado || r.estado || "",
        pdfUrl: r.pdfUrl || r.pdf || "",
        quotes: [],
      });
    });

    // Attach quote rows to requirement items
    quoteRows.forEach(function (q) {
      const rawOrderId = clean(q.orderId || q.refCotiz || q.order || "");
      if (!rawOrderId) return;
      // Consolidated quotes carry combined refs ("A+B"): attach the quote row
      // to EVERY source requirement so the lifecycle report stays accurate.
      splitCombinedRef_(rawOrderId).forEach(function (orderId) {
      // ensure reqMap entry
      reqMap[orderId] = reqMap[orderId] || {
        orderId: orderId,
        createdDate: "",
        site: q.faena || "",
        category: q.category || "",
        creator: "",
        items: [],
        quoteIds: new Set(),
        suppliers: new Set(),
        lastUpdated: q.timestampIso || q.timestamp || "",
      };

      const req = reqMap[orderId];
      // record quote id and supplier
      if (q.quoteId) req.quoteIds.add(clean(q.quoteId));
      if (q.supplier) req.suppliers.add(clean(q.supplier));

      // try to match to an existing item by geo or pn
      let matched = null;
      const qGeo = clean(q.geo || q.geoCode || q.geoCode);
      const qPn = clean(q.pn || q.pn || "");
      for (let i = 0; i < req.items.length; i++) {
        const it = req.items[i];
        if (it.geo && qGeo && clean(it.geo) === qGeo) {
          matched = it;
          break;
        }
        if (it.pn && qPn && clean(it.pn) === qPn) {
          matched = it;
          break;
        }
      }

      // if no match, create a synthetic item entry
      if (!matched) {
        matched = { pn: q.pn || "", geo: q.geo || "", desc: q.desc || "", qty: q.cant || 0, estado: "", pdfUrl: q.quoteLink || "", quotes: [] };
        req.items.push(matched);
      }

      // attach quote item
      matched.quotes = matched.quotes || [];
      matched.quotes.push(q);

      // update lastUpdated
      try {
        const qTime = q.timestampIso ? new Date(q.timestampIso) : q.timestamp ? new Date(q.timestamp) : null;
        if (qTime) {
          const ts = qTime.toISOString();
          if (!req.lastUpdated || new Date(req.lastUpdated) < qTime) req.lastUpdated = ts;
        }
      } catch (e) { }
      }); // end splitCombinedRef_ loop
    });

    // Build response entries
    const entries = Object.keys(reqMap).map(function (orderId) {
      const req = reqMap[orderId];
      // STORE detection: orders destined for Store have "Received" as final status
      const isStore = clean(req.site).toLowerCase() === "store";
      // compute per-item last-known status
      let aggregatedStages = new Set();
      let totalQuotedValue = 0;
      req.items.forEach(function (it) {
        let itemStage = "Pending";
        if (it.quotes && it.quotes.length) {
          // find latest quote item by timestamp
          let latest = it.quotes[0];
          let latestTime = latest.timestampIso ? new Date(latest.timestampIso).getTime() : latest.timestamp ? new Date(latest.timestamp).getTime() : 0;
          for (let j = 1; j < it.quotes.length; j++) {
            const cand = it.quotes[j];
            const candTime = cand.timestampIso ? new Date(cand.timestampIso).getTime() : cand.timestamp ? new Date(cand.timestamp).getTime() : 0;
            if (candTime > latestTime) {
              latest = cand;
              latestTime = candTime;
            }
          }
          itemStage = latest.estado || latest.status || itemStage;
          totalQuotedValue += Number(latest.total || latest.total || 0) || 0;
        }
        let norm = normalizeStage(itemStage);
        // For STORE orders, cap the item status at "Received"
        if (isStore && (norm === "Final Received" || norm === "Despatched")) {
          norm = "Received";
        }
        aggregatedStages.add(norm || itemStage || "Pending");
        it.lastKnownStage = norm || itemStage;
      });

      // Determine requirement-level stage by precedence
      const precedence = ["Final Received", "PO Created", "Despatched", "Approved", "Quoted", "Sent", "Pending", "Not Quoted", "Created"];
      let reqStage = "Created";
      for (let p = 0; p < precedence.length; p++) {
        if ([...aggregatedStages].some(s => s === precedence[p])) {
          reqStage = precedence[p];
          break;
        }
      }
      // For STORE orders, cap the requirement-level status at "Received"
      if (isStore && (reqStage === "Final Received" || reqStage === "Despatched")) {
        reqStage = "Received";
      }

      const quoteIdsArr = Array.from(req.quoteIds || []);
      const suppliersArr = Array.from(req.suppliers || []);

      return {
        orderId: req.orderId,
        createdDate: req.createdDate,
        site: req.site,
        category: req.category,
        creator: req.creator,
        items: req.items,
        quotesCount: quoteIdsArr.length,
        quoteIds: quoteIdsArr,
        supplierCount: suppliersArr.length,
        suppliers: suppliersArr,
        totalQuoted: totalQuotedValue,
        stage: reqStage,
        currentStage: reqStage,
        lastUpdated: req.lastUpdated || req.createdDate || "",
      };
    });

    // sort by lastUpdated desc
    entries.sort(function (a, b) {
      const ta = a.lastUpdated ? new Date(a.lastUpdated).getTime() : 0;
      const tb = b.lastUpdated ? new Date(b.lastUpdated).getTime() : 0;
      return tb - ta;
    });

    return entries;
  } catch (e) {
    console.error("Lifecycle aggregator error: " + e.toString());
    return [];
  }
}

function buildHtmlTable(items) {
  // Per-item notes travel with the requirement email (internal routing
  // recipients only). Kept as a column so the layout stays predictable.
  const notesText = function (item) {
    const s = cleanReqNotes_(item && item.itemNotes);
    return s || "&mdash;";
  };

  // Per-item photos attached in the Create Req cart (Drive URLs uploaded when
  // picked). Column renders only when at least one line has photos, mirroring
  // the conditional Notes logic below. `shipment.items` carries `images` as an
  // array of {url,name} (cart) or the comma-separated "Item Images" cell
  // string (re-loaded drafts) — accept both. Stored URLs come from
  // file.getUrl() (".../file/d/FILEID/view..."); thumbnails use the Drive
  // thumbnail endpoint so the photos render inline in the email, each wrapped
  // in a link to the full file for click-through.
  const showImagesCol = (items || []).some(function (it) {
    return reqItemImageUrls_(it).length > 0;
  });
  const imagesCell = function (item) {
    const urls = reqItemImageUrls_(item);
    if (!urls.length) return "&mdash;";
    const thumbs = urls.map(function (u) {
      const id = reqDriveFileId_(u);
      const thumb = id
        ? ("https://drive.google.com/thumbnail?id=" + id + "&sz=w200")
        : u;
      return '<a href="' + u + '" target="_blank" rel="noopener">' +
        '<img src="' + thumb + '" alt="item image" loading="lazy" ' +
        'style="width:64px;height:64px;object-fit:cover;border:1px solid #ddd;border-radius:4px;margin:1px;vertical-align:middle;">' +
        '</a>';
    }).join("");
    return '<span style="white-space:nowrap;">' + thumbs + '</span>';
  };

  let tableRows = items
    .map((item) => {
      const esDesc = String(item.es || "").trim() || "";
      const enDesc = String(item.en || "").trim() || "";
      const descDisplay = esDesc && enDesc
        ? `<strong>${esDesc}</strong><br><span style="font-size: 10px; color: #666;">${enDesc}</span>`
        : `<strong>${esDesc || enDesc || "---"}</strong>`;

      return `
      <tr>
        <td style="border: 1px solid #ddd; padding: 8px;">${item.geo}</td>
        <td style="border: 1px solid #ddd; padding: 8px;">${item.pn}</td>
        <td style="border: 1px solid #ddd; padding: 8px; font-size: 11px;">${descDisplay}</td>
        <td style="border: 1px solid #ddd; padding: 8px; text-align: center;">${item.qty}</td>
        <td style="border: 1px solid #ddd; padding: 8px; font-size: 11px;">${notesText(item)}</td>
        ${showImagesCol ? `<td style="border: 1px solid #ddd; padding: 8px; font-size: 11px;">${imagesCell(item)}</td>` : ""}
      </tr>`;
    })
    .join("");

  return `<table style="border-collapse: collapse; width: 100%; font-family: sans-serif; font-size: 13px;">
            <thead style="background: #f8f9fa;">
              <tr>
                <th style="border: 1px solid #ddd; padding: 8px;">Geo</th>
                <th style="border: 1px solid #ddd; padding: 8px;">P/N</th>
                <th style="border: 1px solid #ddd; padding: 8px;">Description</th>
                <th style="border: 1px solid #ddd; padding: 8px;">Qty</th>
                <th style="border: 1px solid #ddd; padding: 8px;">Notes</th>
                ${showImagesCol ? `<th style="border: 1px solid #ddd; padding: 8px;">Images</th>` : ""}
              </tr>
            </thead>
            <tbody>${tableRows}</tbody>
          </table>`;
}


/**
* Shared logic to handle folder creation and PDF generation for Requirement
*/
function orchestrateRequirementPDF(orderId, shipment) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const ssFile = DriveApp.getFileById(ss.getId());
    const parentFolder = ssFile.getParents().next();

    // 1. Get/Create "Req" folder
    let reqFolder = parentFolder.getFoldersByName("Req").hasNext()
      ? parentFolder.getFoldersByName("Req").next()
      : parentFolder.createFolder("Req");

    // 2. Get/Create Site folder
    let siteFolder = reqFolder.getFoldersByName(shipment.site).hasNext()
      ? reqFolder.getFoldersByName(shipment.site).next()
      : reqFolder.createFolder(shipment.site);

    // 3. Clean up old PDF if it exists (prevents multiple versions of the same Order ID)
    const existingFiles = siteFolder.getFilesByName(orderId + ".pdf");
    while (existingFiles.hasNext()) {
      existingFiles.next().setTrashed(true);
    }

    // 4. Generate the PDF
    return generateRequirementPDF(
      orderId,
      shipment,
      new Date().toLocaleString(),
      siteFolder
    );
  } catch (e) {
    console.error("Orchestration failed for " + orderId + ": " + e.toString());
    return ""; // Return empty string so the spreadsheet doesn't crash
  }
}

/**
 * Parse an Ordenes "Fecha" cell into a Date.
 *
 * The main creation flow (CreateReqGS) writes a real Date object, but the legacy
 * mass-orders path writes the stamp as text in "dd/MM/yyyy HH:mm" form. Handing
 * that string to new Date() would read it as US M/D (and fail outright for days
 * > 12), so the day-first pattern is parsed explicitly.
 */
function parseRequirementDate_(value) {
  if (value === null || value === undefined) return null;
  if (Object.prototype.toString.call(value) === "[object Date]") {
    return isNaN(value.getTime()) ? null : value;
  }
  const raw = String(value).trim();
  if (!raw) return null;
  const dmy = raw.match(/^(\d{1,2})[\/.](\d{1,2})[\/.](\d{4})(?:[ ,T]+(\d{1,2}):(\d{2}))?/);
  if (dmy) {
    const asDate = new Date(Number(dmy[3]), Number(dmy[2]) - 1, Number(dmy[1]), Number(dmy[4] || 0), Number(dmy[5] || 0));
    return isNaN(asDate.getTime()) ? null : asDate;
  }
  const parsed = new Date(raw);
  return isNaN(parsed.getTime()) ? null : parsed;
}

function getRequirementsHistory() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Ordenes");

    if (!sheet) return [];

    const data = sheet.getDataRange().getValues();

    if (data.length <= 1) return [];

    // Tag each row with its physical sheet row number (1-based; row 1 is the
    // header) BEFORE filtering, so the edit modal can pair each item back to
    // the exact row on save. The history is returned newest-first below, so
    // the server must NOT pair items to rows by array index (it used to,
    // which wrote each item's Geo/P/N/Desc onto the wrong row).
    const cleanData = data
      .slice(1)
      .map((row, idx) => ({ row: row, rowNum: idx + 2 }))
      .filter((entry) => entry.row[0] !== "" && entry.row[0] !== null)
      .filter((entry) => {
        // Exclude drafts from history view
        const estado = String(entry.row[7] || "").trim();
        return estado !== "Draft" && !String(entry.row[0] || "").trim().startsWith("DRAFT-");
      });

    // Header-aware lookup of the optional "SubArea" / "Emails" / "SubArea No"
    // and "Notes" columns, plus the requirement AUDIT columns
    // ("lastModifiedBy" / "lastModifiedDate" / "modCount") written by CreateReqGS
    // at creation and by every requirement edit since. The audit headers are
    // matched case-insensitively AND with the spaced spelling ("Last Modified By"),
    // because the sheet has existed in both forms; a -1 index simply means the
    // column is missing (legacy sheet / never edited) and the consumer shows "-".
    const headers = data[0];
    let subAreaIdx = -1;
    let emailsIdx = -1;
    let subAreaNoIdx = -1;
    let notesIdx = -1;
    let itemNotesIdx = -1;
    let supplierNotesIdx = -1;
    let itemImagesIdx = -1;
    let lastModByIdx = -1;
    let lastModDateIdx = -1;
    let modCountIdx = -1;
    for (let i = 0; i < headers.length; i++) {
      const headerName = String(headers[i] || "").trim();
      const headerKey = headerName.toLowerCase();
      if (headerName === "SubArea") subAreaIdx = i;
      if (headerName === "Emails") emailsIdx = i;
      if (headerName === "SubArea No") subAreaNoIdx = i;
      if (headerName === "Notes") notesIdx = i;
      if (headerKey === "item notes") itemNotesIdx = i;
      if (headerKey === "supplier notes") supplierNotesIdx = i;
      if (headerKey === "item images") itemImagesIdx = i;
      if (headerKey === "lastmodifiedby" || headerKey === "last modified by") lastModByIdx = i;
      if (headerKey === "lastmodifieddate" || headerKey === "last modified date") lastModDateIdx = i;
      if (headerKey === "modcount" || headerKey === "mod count") modCountIdx = i;
    }

    // Requirement audit date cells arrive either as a real Date (CreateReqGS
    // stamps new Date()) or as the "yyyy-MM-dd HH:mm:ss" text ViewEditReqGS
    // writes. Both are normalized to { text, iso }: `text` is the display value
    // (dd-MM-yyyy HH:mm in the script timezone, or the original string when it
    // cannot be parsed) and `iso` is a sortable timestamp for the client.
    const auditDate_ = function (v) {
      if (Object.prototype.toString.call(v) === "[object Date]") {
        if (isNaN(v.getTime())) return { text: "", iso: "" };
        return {
          text: Utilities.formatDate(v, Session.getScriptTimeZone(), "dd-MM-yyyy HH:mm"),
          iso: v.toISOString(),
        };
      }
      const raw = String(v == null ? "" : v).trim();
      if (!raw) return { text: "", iso: "" };
      const parsed = new Date(raw);
      if (isNaN(parsed.getTime())) return { text: raw, iso: "" };
      return {
        text: Utilities.formatDate(parsed, Session.getScriptTimeZone(), "dd-MM-yyyy HH:mm"),
        iso: parsed.toISOString(),
      };
    };

    return cleanData
      .map((entry) => {
        const row = entry.row;
        // Machine-readable requirement date next to the locale string: consumers
        // that must sort or format the Req date (e.g. the Quote Mgmt requirement
        // header) cannot rely on toLocaleString(), which is locale dependent.
        const reqDate = parseRequirementDate_(row[1]);
        const reqDateValid = !!reqDate;
        // Single parse of the requirement audit date (text + iso computed once).
        const reqLastMod = lastModDateIdx >= 0
          ? auditDate_(row[lastModDateIdx])
          : { text: "", iso: "" };
        return {
          rowNum: entry.rowNum,
          order: row[0] || "N/A",
          fecha: row[1] ? row[1].toLocaleString() : "",
          fechaIso: reqDateValid ? reqDate.toISOString() : "",
          pn: row[2] || "",
          geo: row[3] || "",
          faena: row[4] || "",
          desc: row[5] || "",
          cant: row[6] || 0,
          estado: row[7] || "Pending",
          user: row[8],
          category: row[12] || "General",
          pdfUrl: row[13] || "",
          subArea: subAreaIdx >= 0 ? (row[subAreaIdx] || "") : "",
          emails: emailsIdx >= 0 ? (row[emailsIdx] || "") : "",
          subAreaNo: subAreaNoIdx >= 0 ? (row[subAreaNoIdx] || "") : "",
          notes: notesIdx >= 0 ? (row[notesIdx] || "") : "",
          // Per-item notes (unique to this requirement row: Order + Geo)
          itemNotes: itemNotesIdx >= 0 ? (row[itemNotesIdx] || "") : "",
          // Supplier-facing notes ("Supplier Notes" column): printed in the
          // supplier RFQ (quote request) email only.
          supplierNotes: supplierNotesIdx >= 0 ? (row[supplierNotesIdx] || "") : "",
          // Per-item photos ("Item Images" column): comma-separated Drive URLs
          // uploaded from the Create Req cart. Returned as the raw cell string;
          // consumers normalize with reqItemImageUrls_ / normalizeCartImages.
          images: itemImagesIdx >= 0 ? (row[itemImagesIdx] || "") : "",
          // Requirement AUDIT trail (last requirement edit). Joined onto the
          // Quote Mgmt requirement header's info ("i") timeline: an untouched
          // requirement reports its creation values (CreateReqGS stamps
          // lastModifiedDate at creation; lastModifiedBy stays blank until the
          // first edit) and a legacy sheet without these columns reports "-".
          reqLastModifiedBy: lastModByIdx >= 0 ? String(row[lastModByIdx] || "").trim() : "",
          reqLastModified: reqLastMod.text,
          reqLastModifiedIso: reqLastMod.iso,
          reqModCount: modCountIdx >= 0 ? (Number(row[modCountIdx]) || 0) : 0,
        };
      })
      .reverse();
  }

function getLiveStockMap(site) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const stockSheet = ss.getSheetByName("Stock");
  const stockMap = {};
  
  if (!stockSheet) return stockMap;

  const data = stockSheet.getDataRange().getValues();
  const hasSiteFilter = site && String(site).trim() !== "";
  const normalizedSite = hasSiteFilter ? normalizeLocationId_(site) : "";
  const canonicalType = normalizeLocationType_(LOCATION_TYPE_SECTOR);

  // Loop through rows skipping the header row
  for (let i = 1; i < data.length; i++) {
    const itemId = normalizeLocationId_(data[i][0]);   // ItemID (Column A)
    if (!itemId) continue;

    const locType = normalizeLocationType_(data[i][2]); // LocationType (Column C)
    const locationId = normalizeLocationId_(data[i][3]); // LocationID / Sector name (Column D)
    const available = Number(data[i][6]) || 0;           // AvailableQty (Column G)

    if (hasSiteFilter) {
      // Rule: If the Site is not empty, use the stock of said Site
      if (locationId === normalizedSite) {
        stockMap[itemId] = (stockMap[itemId] || 0) + available;
      }
    } else {
      // Fallback: no site selected yet (Admin/Office view) -> every sector.
      // All locations are sectors, so this is simply the canonical type.
      if (locType === canonicalType) {
        stockMap[itemId] = (stockMap[itemId] || 0) + available;
      }
    }
  }

  return stockMap;
}

function getMinMaxData() {
  // Global fallback values from Master sheet.
  const globalByGeo = new Map();
  try {
    (getInventoryData() || []).forEach((i) => {
      if (i.geo) {
        globalByGeo.set(String(i.geo).trim(), { min: i.min, max: i.max });
      }
    });
  } catch (e) {
    Logger.log("getMinMaxData: inventory fallback error " + e);
  }

  // Sites list from Validation.
  let sites = [];
  try {
    const val = getValidationData();
    sites = (val.sites || []).map((s) => s.name);
  } catch (e) {
    Logger.log("getMinMaxData: sites error " + e);
  }

  const sheet = getMinMaxSheet_();
  const vals = sheet.getDataRange().getValues();
  const rows = [];
  for (let i = 1; i < vals.length; i++) {
    const site = String(vals[i][0] || "").trim();
    const geo = String(vals[i][1] || "").trim();
    if (!site || !geo) continue;
    rows.push({
      site,
      geo,
      min: Number(vals[i][2]) || 0,
      max: Number(vals[i][3]) || 0,
      lastUpdated: vals[i][4] ? String(vals[i][4]) : "",
      updatedBy: String(vals[i][5] || "") || "",
    });
  }

  // Products with global fallback (for display / defaults).
  const products = Array.from(globalByGeo.entries()).map(([geo, fb]) => ({
    geo,
    globalMin: fb.min,
    globalMax: fb.max,
  }));

  return { rows, products, sites };
}

function getMovementValidation() {

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Movement_Val");

  if (!sheet) {
    throw new Error("Movement_Val sheet not found");
  }

  const data = sheet.getDataRange().getValues();

  if (data.length < 2) {
    return {
      Sector: [],
      Maquina: [],
      Generador: [],
      Vehiculo: []
    };
  }

  const headers = data[0];

  const colIndex = {
    Sector: headers.indexOf("Sector"),
    Maquina: headers.indexOf("Maquina"),
    Generador: headers.indexOf("Generador"),
    Vehiculo: headers.indexOf("Vehiculo")
  };

  const result = {
    Sector: new Set(),
    Maquina: new Set(),
    Generador: new Set(),
    Vehiculo: new Set()
  };

  for (let i = 1; i < data.length; i++) {

    const row = data[i];

    if (colIndex.Sector !== -1 && row[colIndex.Sector])
      result.Sector.add(String(row[colIndex.Sector]).trim());

    if (colIndex.Maquina !== -1 && row[colIndex.Maquina])
      result.Maquina.add(String(row[colIndex.Maquina]).trim());

    if (colIndex.Generador !== -1 && row[colIndex.Generador])
      result.Generador.add(String(row[colIndex.Generador]).trim());

    if (colIndex.Vehiculo !== -1 && row[colIndex.Vehiculo])
      result.Vehiculo.add(String(row[colIndex.Vehiculo]).trim());

  }

  return {
    Sector: [...result.Sector],
    Maquina: [...result.Maquina],
    Generador: [...result.Generador],
    Vehiculo: [...result.Vehiculo]
  };
}

===== getPendingRequirementsForRFQ (Código.js) =====
function getPendingRequirementsForRFQ(site, statusFilter) {
  const allHistory = getRequirementsHistory();
  const masterData = getInventoryData();

  const masterMap = new Map();
  masterData.forEach((item) => {
    // We add 'supplier' to the map retrieval
    masterMap.set(item.geo, {
      en: item.en,
      es: item.es,
      supplier: item.proveedor,
      category: item.category,
      subCategory: item.sub_categoria,
    });
  });

  // statusFilter: "Approved" (default) | "All" (everything except Pending, i.e.
  // requirements still awaiting approval are excluded) | any specific status
  // word (Quoted, Sent, Received, Despatched, ...).
  const statusKey = String(statusFilter || "Approved").trim().toLowerCase();
  let pending;
  if (statusKey === "all") {
    pending = allHistory.filter(
      (item) =>
        item.estado &&
        !item.estado.toString().toLowerCase().includes("pend"),
    );
  } else {
    pending = allHistory.filter(
      (item) =>
        item.estado && item.estado.toString().toLowerCase().includes(statusKey),
    );
  }
  if (site) {
    pending = pending.filter((item) => item.faena === site);
  }

  // Existing quotes per requirement item (for the "additional quote" flow):
  // lets the UI show which suppliers already quoted each GEO and block
  // re-selecting the same supplier for the same item.
  const quotesByOrderGeo = {};
  try {
    (getQuotesHistory() || []).forEach(function (q) {
      if (String(q.estado || "").toLowerCase() === "deleted") return;
      const key = String(q.orderId || "").trim() + "||" + String(q.geo || "").trim();
      (quotesByOrderGeo[key] = quotesByOrderGeo[key] || []).push({
        supplier: q.supplier || "",
        quoteId: q.quoteId || "",
        status: q.estado || "",
      });
    });
  } catch (e) {
    console.warn("RFQ quote-info join failed: " + e);
  }

  const groupedReqs = {};
  pending.forEach((item) => {
    if (!groupedReqs[item.order]) {
      groupedReqs[item.order] = {
        id: item.order,
        date: item.fecha,
        site: item.faena,
        subArea: item.subArea || "",
        status: item.estado || "",
        items: [],
      };
    }

    const official = masterMap.get(item.geo) || {
      en: "",
      es: "",
      supplier: "",
      category: "",
      subCategory: "",
    };
    groupedReqs[item.order].items.push({
      p: item.pn || "",
      geo: item.geo || "",
      // Capture both descriptions clearly
      es: (official.es || "").toString().trim(),
      en: (official.en || "").toString().trim(),
      // Keep the raw requirement description as a safety reference
      req_desc: (item.desc || "").toString().trim(),
      qty: item.cant || 0,
      supplier: official.supplier || "",
      category: resolveRequirementCategory_(item.category, official.category),
      subCategory: official.subCategory || "",
      quotedBy: quotesByOrderGeo[
        String(item.order || "").trim() + "||" + String(item.geo || "").trim()
      ] || [],
      // Item notes are unique to each requirement, so the Req ID travels with
      // the note: consolidated rows can render "Req 100123: <note>".
      reqItemNotes: [
        {
          ref: String(item.order || "").trim(),
          notes: String(item.itemNotes || ""),
        },
      ],
      // Supplier-facing notes travel the same way. These are the ONLY per-item
      // notes printed in the supplier RFQ email ("Item Notes" stays internal).
      reqSupplierNotes: [
        {
          ref: String(item.order || "").trim(),
          notes: String(item.supplierNotes || ""),
        },
      ],
      // Per-item photos (Ordenes "Item Images": comma-separated Drive URLs).
      // Kept per source Req (raw cell string) so the RFQ table, the preview and
      // the supplier email can render the requester's attachments. The client
      // unions them when several requirements are consolidated onto one GEO.
      images: String(item.images || ""),
    });
  });

  return Object.values(groupedReqs);
}

/**

 * Renders single or dual tables depending on showDeficit/showInStock flags

function buildRFQEmailBody_(baseBody, deficitItems, inStockItems, options) {
  options = options || {};
  const showDeficit = options.showDeficit !== false;
  const showInStock = options.showInStock !== false;

  let emailHtml = String(baseBody || "").replace(/\n/g, "<br>");

  // ---- Supplier Notes (per item) -----------------------------------------
  // "Supplier Notes" is typed per item in the Create Req cart and stored on the
  // Ordenes "Supplier Notes" column. It is the ONLY per-item note a supplier may
  // see ("Item Notes" stays internal). The column is rendered only when at least
  // one item carries a note, so emails without supplier notes stay unchanged.
  const supplierNotesText = function (item) {
    if (!item) return "";
    const parts = [];
    if (Array.isArray(item.reqSupplierNotes)) {
      item.reqSupplierNotes.forEach(function (n) {
        const note = cleanReqNotes_(n && n.notes);
        if (!note) return;
        const ref = String((n && n.ref) || "").trim();
        parts.push(ref ? ref + ": " + note : note);
      });
    }
    if (!parts.length) {
      const single = cleanReqNotes_(item.supplierNotes);
      if (single) parts.push(single);
    }
    return parts.join(" | ");
  };
  const hasSupplierNotes = function (items) {
    return (items || []).some(function (it) { return supplierNotesText(it) !== ""; });
  };
  const supplierNotesThHtml = '<th style="padding:8px; border:1px solid #999; text-align:left;">Notas / Notes (Proveedor)</th>';
  const showSupplierNotesDeficit = showDeficit && hasSupplierNotes(deficitItems);
  // NOTE: the in-stock table is only ever sent to INTERNAL users (scenarios A/B),
  // so "Supplier Notes" is deliberately not added there.

  // Per-item photos attached in the Create Req cart (Ordenes "Item Images").
  // Rendered as linked Drive thumbnails in the deficit (supplier-facing) table
  // and in the internal in-stock table, but ONLY when at least one line carries
  // a photo — imageless RFQs keep byte-identical bodies. Files are link-shared
  // view-only at upload time, so the supplier can open the full photo.
  const imagesThHtml = '<th style="padding:8px; border:1px solid #999; text-align:left;">Images</th>';
  const showImagesDeficit = showDeficit && (deficitItems || []).some(function (it) {
    return reqItemImageUrls_(it).length > 0;
  });
  const showImagesInStock = showInStock && (inStockItems || []).some(function (it) {
    return reqItemImageUrls_(it).length > 0;
  });

  // Render Deficit Items table (if applicable)
  if (showDeficit && deficitItems && deficitItems.length > 0) {
    const deficitTableHtml = `
      <br><br>
      <h3 style="color:#d9534f; font-size:14px; font-weight:bold;">Items Requiring Purchase</h3>
      <table style="width:100%; border-collapse:collapse; margin:15px 0; font-size:12px; font-family: Arial, sans-serif;">
        <tr style="background:#f0f0f0; border:1px solid #999;">
          <th style="padding:8px; border:1px solid #999; text-align:left;">GEO Code</th>
          <th style="padding:8px; border:1px solid #999; text-align:left;">P/N</th>
          <th style="padding:8px; border:1px solid #999; text-align:left;">Descripción / Description</th>
          <th style="padding:8px; border:1px solid #999; text-align:center;">Cant.</th>
          ${showSupplierNotesDeficit ? supplierNotesThHtml : ""}
          ${showImagesDeficit ? imagesThHtml : ""}
        </tr>
        ${deficitItems
        .map(
          (item) => `<tr style="border:1px solid #999;">
          <td style="padding:6px; border:1px solid #999;">${item.geo}</td>
          <td style="padding:6px; border:1px solid #999;"><strong>${item.pn || item.p || ""}</strong></td>
          <td style="padding:6px; border:1px solid #999;">
            <div style="font-weight:bold; color:#000;">${item.es || item.req_desc || "N/A"}</div>
            <div style="color:#666; font-size:11px; font-style: italic;">${item.en || ""}</div>
          </td>
          <td style="padding:6px; border:1px solid #999; text-align:center;"><strong>${item.qty}</strong></td>
          ${showSupplierNotesDeficit ? '<td style="padding:6px; border:1px solid #999; font-size:11px;">' + supplierNotesText(item) + '</td>' : ""}
          ${showImagesDeficit ? '<td style="padding:6px; border:1px solid #999; font-size:11px;">' + reqImagesCellHtml_(item, 56) + '</td>' : ""}
        </tr>`,
        )
        .join("")}
      </table>`;
    emailHtml += deficitTableHtml;
  }

  // Render In-Stock Items table (if applicable)
  if (showInStock && inStockItems && inStockItems.length > 0) {
    const inStockTableHtml = `
      <br><br>
      <h3 style="color:#5cb85c; font-size:14px; font-weight:bold;">Items Available in Stock</h3>
      <table style="width:100%; border-collapse:collapse; margin:15px 0; font-size:12px; font-family: Arial, sans-serif;">
        <tr style="background:#f0f0f0; border:1px solid #999;">
          <th style="padding:8px; border:1px solid #999; text-align:left;">GEO Code</th>
          <th style="padding:8px; border:1px solid #999; text-align:left;">P/N</th>
          <th style="padding:8px; border:1px solid #999; text-align:left;">Descripción / Description</th>
          <th style="padding:8px; border:1px solid #999; text-align:center;">Cant.</th>
          ${showImagesInStock ? imagesThHtml : ""}
        </tr>
        ${inStockItems
        .map(
          (item) => `<tr style="border:1px solid #999;">
          <td style="padding:6px; border:1px solid #999;">${item.geo}</td>
          <td style="padding:6px; border:1px solid #999;"><strong>${item.pn || item.p || ""}</strong></td>

        </tr>`,
        )
        .join("")}
      </table>`;
    emailHtml += inStockTableHtml;
  }

  return emailHtml;
}

/**
 * Modify email body greeting from supplier-specific to generic for users
 * Replaces "Estimado [name]," at start with "Equipo,"
 */
function personalizeBodyForUsers_(body) {
  if (!body) return body;
  // Replace "Estimado [anything]," at start with "Equipo,"
  return String(body).replace(/^([\s]*Estimado\s+[^,]*,)/i, "Equipo,");
}

/**
 * Splits a combined Ref ID ("100123+100127") into its source requirement IDs.
 * Plain single refs pass through unchanged (returns [ref]).
 */
function splitCombinedRef_(refId) {
  return String(refId || "")
    .split("+")
    .map(function (id) { return String(id || "").trim(); })
    .filter(Boolean);
}

function sendFinalRFQ(details) {

  try {
    
  const flags = getQuoteConfigFlags_();

  // =====================================================
  // PHASE 2: EXTRACT AND CLASSIFY ITEMS
  // =====================================================
  const deficitItems = details.deficitItems || [];
  const bodegaItems = details.bodegaItems || [];
  const hasInStockItems = deficitItems.length === 0 ? false : (bodegaItems.length > 0);

  console.log("PHASE 2 - Stock Classification:", {
    deficitCount: deficitItems.length,
    bodegaCount: bodegaItems.length,
    hasInStockItems: hasInStockItems
  });

  // Determine scenario:
  // Scenario A: Mixed inventory (both deficit & in-stock)
  // Scenario B: All in-stock (zero deficit)
  // Scenario C: All out of stock (zero in-stock)
  const isScenarioA = deficitItems.length > 0 && bodegaItems.length > 0;
  const isScenarioB = deficitItems.length === 0 && bodegaItems.length > 0;
  const isScenarioC = deficitItems.length > 0 && bodegaItems.length === 0;

  console.log("PHASE 2 - Scenario:", {
    isScenarioA: isScenarioA,
    isScenarioB: isScenarioB,
    isScenarioC: isScenarioC
  });

  // Use deficit items for PDF generation (base quoteItems).
  // 🛎️ FIX (PDF_ONLY mode): details.deficitItems / details.bodegaItems are only
  // populated by the EMAIL_AND_PDF payload - the PDF_ONLY payload never sent
  // them, so both were []. `bodegaItems` (an empty array) was then handed to
  // buildQuoteItems_() and, being truthy, it blanked out details.items, making
  // every PDF_ONLY send fail with "No supplier items were provided for this RFQ".
  // Fall through to details.items so any payload shape still yields a quote.
  const quoteItems = buildQuoteItems_(
    details,
    deficitItems.length > 0
      ? deficitItems
      : (bodegaItems.length > 0 ? bodegaItems : (details.items || []))
  );

  if (!quoteItems.length) {
    return {
      success: false,
      message: "No supplier items were provided for this RFQ.",
    };
  }

  // 🔒 FIX: CC routing must reflect the REAL categories in this quote, never a
  // fallback "General". Collect the distinct categories across ALL items involved
  // (deficit + in-stock) so category staff for every product are CC'd.
  const routingCategories = Array.from(new Set(
    (deficitItems.concat(bodegaItems))
      .map(function (it) { return String(it.category || "").trim(); })
      .filter(Boolean)
  )).sort();
  const routingCategoriesText = routingCategories.length > 0
    ? routingCategories.join(", ")
    : String(details.category || "").trim();

  const orderId = String(details.orderId || details.refId || "PENDING");
  console.log("first 2 sendfinalrfq, ", details.orderId);

  // Consolidated RFQ support: details.orderIds carries every source requirement
  // ("100123+100127" style combined refs are split as a fallback). Single-req
  // RFQs simply get a one-element list.
  const orderIds = (Array.isArray(details.orderIds) && details.orderIds.length)
    ? details.orderIds.map(function (id) { return String(id || "").trim(); }).filter(Boolean)
    : splitCombinedRef_(orderId);
  // Ref ID written on the Quotes sheet: combined when the quote consolidates
  // several requirements, plain single ref otherwise.
  const refIdForQuote = orderIds.length > 1 ? orderIds.join("+") : (orderIds[0] || orderId);

  // Persist any supplier notes the user edited in the RFQ items table. Only items
  // that explicitly carry `supplierNotes` were touched in the UI (see the client
  // setRfqSupplierNotes), so untouched requirements are never rewritten and the
  // "Supplier Notes" column is only created when a note is actually set. Runs on
  // SEND only (this function), matching the "save on send" contract.
  try {
    persistRfqSupplierNotes_(details, orderIds);
  } catch (noteErr) {
    console.warn("sendFinalRFQ: supplier notes persist failed: " + noteErr.toString());
  }

  const quoteId = String(
    details.quoteId ||
    `RFQ-${orderId}-${Utilities.formatDate(new Date(), "GMT-4", "yyyyMMdd-HHmmss")}-${Utilities.getUuid()}`,
  );

  const supplierInfo = getQuoteSupplierInfo_(String(details.supplier || ""));

  const quoteData = {
    orderId: refIdForQuote,
    // Consolidated RFQ traceability: list the source requirements in Comments
    orderIds: orderIds,
    quoteId: quoteId,
    quoteFileName: sanitizeFileName_(quoteId),
    supplier: String(details.supplier || "N/A"),
    site: String(details.site || ""),
    category: String(details.category || "General"),
    comprador: String(
      details.userName || details.comprador || details.replyTo || "",
    ),
    createdBy: String(details.currentUserEmail || details.replyTo || Session.getActiveUser().getEmail() || ""),
    createdDate: new Date(),
    lastModifiedBy: String(details.currentUserEmail || details.replyTo || Session.getActiveUser().getEmail() || ""),
    lastModifiedDate: new Date(),
    currency: "CLP",
    comments: String(details.comments || ""),
    condicionPago: String(
      details.condicionPago || supplierInfo.paymentTerms || "",
    ),
    ivaInc: String(details.ivaInc || "Neto"),
    refCotiz: "",
    status: "Pending",
    items: quoteItems,
  };

  // Traceability: consolidated quotes record their source requirements in Comments
  if (orderIds.length > 1) {
    quoteData.comments = ("[Consolidated Reqs: " + orderIds.join("+") + "] " + quoteData.comments).trim();
  }

  console.log("STEP 1 - items before pricing:", JSON.stringify(quoteData.items));

  quoteData.items = quoteData.items.map(function (item, i) {
    try {
      const qty = Number(item.qty || item.cant || 0);
      const geo = item.geo || "";
      const pn = item.pn || item.p || "";

      let price = 0;

      if (geo) {
        try {
          price = Number(getQuoteLastPriceFromCompras_(geo, pn) || 0);
        } catch (e) {
          console.warn("Price fetch failed:", geo, pn, e);
          price = 0;
        }
      }

      return {
        ...item,
        qty: qty,
        pn: pn,
        price: price,
        total: price * qty,
        lineTotal: price * qty,
      };
    } catch (err) {
      console.error("Item processing failed:", item, err);
      return {
        ...item,
        price: 0,
        total: 0,
      };
    }
  });

  const targetFolder = getQuoteTargetFolder_(
    quoteData.site,
    quoteData.supplier
  );

  console.log("PDF ITEMS:", JSON.stringify(quoteData.items));

  // =====================================================
  // PHASE 4: PDF GENERATION (CONDITIONAL)
  // =====================================================
  // PDF is generated for Scenarios A (users), B (users), and C (supplier+users)
  // PDF contains all items (deficit + in-stock combined)
  let pdfUrl = null;
  let pdfLinkHtml = "";

  // Only generate PDF if email is enabled
  if (flags.quoEmailEnabled === true) {
    pdfUrl = generateQuotePDF(quoteData, targetFolder);

    if (!pdfUrl || typeof pdfUrl !== "string" || pdfUrl.trim() === "") {
      throw new Error("PDF generation failed - invalid or empty URL returned");
    }

    pdfLinkHtml = `<br><br><b>PDF:</b> <a href="${pdfUrl}" target="_blank">View Quote</a>`;
  }

  // =====================================================
  // SAVE TO SHEET (PROTECTED)
  // =====================================================
  // 🔒 FIX #4: Persist EVERY email involved in this quote creation (supplier TO
  // + internal CC + category routing list) so that later editing / updation emails
  // for the same quote reuse the exact same distribution list.

  const toRecipients = String(details.to || details.initialRecipientEmails || details.replyTo || details.currentUserEmail || '').trim();
  const ccRecipients = String(details.cc || '').trim();
  let routingEmailsForQuote = "";
  try {
    const routingForQuote = getRequirementRouting(details.site || '', routingCategoriesText, String(details.subArea || ''));
    if (routingForQuote && routingForQuote.emails && routingForQuote.emails !== 'procurement@yourcompany.com') {
      routingEmailsForQuote = routingForQuote.emails;
    }
  } catch (err) {
    console.warn('Routing lookup failed while persisting quote recipients: ' + err.toString());
  }
  // Determine which emails to save in the sheet:
  // - If supplierEmail is explicitly provided, exclude it from saved emails
  // - Only save CC + routing emails (these are internal distribution lists)
  const supplierEmailToExclude = String(details.supplierEmail || '').trim().toLowerCase();
  let emailsForSheet = mergeEmailRecipients_([ccRecipients, routingEmailsForQuote]);
  
  // Remove supplier email from saved list if present (defensive check)
  if (supplierEmailToExclude) {
    const emailList = emailsForSheet.split(',').map(e => e.trim().toLowerCase()).filter(Boolean);
    const filteredEmails = emailList.filter(e => e !== supplierEmailToExclude);
    emailsForSheet = filteredEmails.join(', ');
  }
  
  quoteData.initialRecipients = emailsForSheet;

  quoteData.ccEmails = ccRecipients;  // kept for reference / future use
  // Pre-uploaded quote document (attached from the RFQ preview card) is stamped
  // onto the newly created quote rows ("Quoted Doc Link" column).
  const rfqAttachmentUrl = String(details.attachmentUrl || "").trim();
  if (rfqAttachmentUrl) {
    quoteData.quotedDocUrl = rfqAttachmentUrl;
  }
  // 🛎️ FIX (duplicate quote blocks): saveQuoteToSheet() used to run HERE, before
  // any send had actually happened. Every failure downstream - the PDF_ONLY
  // throw below, a missing user-email list, a SmartDispatcher error, or the
  // outer catch - therefore left a complete block of orphan rows behind, and the
  // user's retry appended a SECOND identical block under a brand-new Quote ID
  // (timestamp + UUID). That is precisely the duplicated 16-row blocks seen on
  // the Quotes sheet.
  //
  // The write is now DEFERRED and executed exactly once, on a success path,
  // immediately before updateQuoteSheetStatus_() stamps those rows.
  let quotePersisted = false;
  const persistQuote_ = function () {
    if (quotePersisted) return;
    quotePersisted = true;
    saveQuoteToSheet(quoteData, pdfUrl, emailsForSheet);
  };

  const emailEnabled = flags.quoEmailEnabled === true;

  // =====================================================
  // PDF ONLY FLOW
  // =====================================================
  if (details.mode === "PDF_ONLY") {
    if (!pdfUrl) {
      throw new Error("PDF_ONLY mode failed: PDF not generated");
    }

    persistQuote_();
    updateQuoteSheetStatus_(quoteId, "Pending");

    const itemGeos =
      details.itemGeos ||
      quoteData.items.map(function (item) {
        return item.geo;
      });

    updateQuotedStatus(orderId, quoteId, itemGeos);
    updateOrdenesWithQuote(orderId, quoteId, itemGeos);

    return {
      success: true,
      service: "PDF",
      message: "PDF generated successfully. Email is Disabled by system",
      pdfUrl: pdfUrl,
    };
  }

  // =====================================================
  // EMAIL DISABLED FLOW
  // =====================================================
  if (!emailEnabled) {
    persistQuote_();
    updateQuoteSheetStatus_(quoteId, "Pending");

    const itemGeos =
      details.itemGeos ||
      quoteData.items.map(function (item) {
        return item.geo;
      });

    updateQuotedStatus(orderId, quoteId, itemGeos);
    updateOrdenesWithQuote(orderId, quoteId, itemGeos);
    return {
      success: true,
      service: "Saved",
      message: "Quote saved. Email disabled by Config.",
      pdfUrl: pdfUrl,
    };
  }

  // =====================================================
  // EMAIL FLOW - PHASE 5: CONDITIONAL EMAIL DISPATCH
  // =====================================================
  const senderEmail = String(
    details.currentUserEmail || details.replyTo || Session.getActiveUser().getEmail() || "",
  ).trim();
  let signatureHtml = "";
  if (senderEmail) {
    try {
      signatureHtml = getUserSignature(senderEmail);
    } catch (sigErr) {
      console.warn("RFQ signature lookup failed for " + senderEmail + ": " + sigErr.toString());
    }
  }

  const baseBody = String(details.body || "");
  const subject = String(details.subject || "No Subject");

  // Scenario A: MIXED INVENTORY - Send split emails
  if (isScenarioA) {
    console.log("SCENARIO A: Mixed inventory - sending split emails");

    // EMAIL 1: Supplier (deficit items only, NO PDF)
    const supplierEmailBody = buildRFQEmailBody_(baseBody, deficitItems, [], {
      showDeficit: true,
      showInStock: false
    });

    const supplierEmailPackage = {
      type: "RFQ",
      refId: String(details.orderId || "PENDING"),
      to: String(details.to || ""),
      cc: "", // No CC for supplier in split scenario
      subject: subject,
      body: supplierEmailBody + (signatureHtml ? `<br><br>${signatureHtml}` : ""),
      supplier: String(details.supplier || "N/A"),
      replyTo: String(details.replyTo || Session.getActiveUser().getEmail()),
      senderEmail: senderEmail,
      fallbackProvider: String(details.fallbackProvider || ""),
      pdfUrl: "" // NO PDF for supplier
    };

    console.log("Dispatching Supplier Email (Scenario A)...");
    const supplierResult = SmartDispatcher(supplierEmailPackage);

    // EMAIL 2: Users (all items - deficit + in-stock, WITH PDF)
    const routing = getRequirementRouting(details.site || "", routingCategoriesText, String(details.subArea || ""));
    let userEmails = routing.emails || "";

    // If routing returned fallback, use CC field from form instead
    if (userEmails === "procurement@yourcompany.com" || userEmails.trim() === "") {
      userEmails = String(details.cc || "").trim();
    }

    // If still empty, abort and ask user
    if (!userEmails) {
      return {
        success: false,
        message: "No user emails found from Users sheet. Please add emails in the CC field and try again."
      };
    }

    const personalizedBaseBody = personalizeBodyForUsers_(baseBody);
    const userEmailBody = buildRFQEmailBody_(personalizedBaseBody, deficitItems, bodegaItems, {
      showDeficit: true,
      showInStock: true
    });

    const userEmailPackage = {
      type: "RFQ",
      refId: String(details.orderId || "PENDING"),
      to: userEmails,
      cc: "", // No CC for users in split scenario
      subject: subject + " [Stock Available]",
      body: userEmailBody + pdfLinkHtml + (signatureHtml ? `<br><br>${signatureHtml}` : ""),
      supplier: String(details.supplier || "N/A"),
      replyTo: String(details.replyTo || Session.getActiveUser().getEmail()),
      senderEmail: senderEmail,
      fallbackProvider: String(details.fallbackProvider || ""),
      pdfUrl: pdfUrl
    };

    console.log("Dispatching User Email (Scenario A)...");
    const userResult = SmartDispatcher(userEmailPackage);

    // Log both emails and update status
    if (supplierResult && supplierResult.success) {
      console.log("Supplier email sent successfully");
    } else {
      console.warn("Supplier email dispatch returned:", supplierResult);
    }

    if (userResult && userResult.success) {
      console.log("User email sent successfully");
      persistQuote_();
      updateQuoteSheetStatus_(quoteId, "Pending");

      const itemGeos = details.itemGeos || quoteData.items.map(function (item) {
        return item.geo;
      });

      updateQuotedStatus(orderId, quoteId, itemGeos);
      updateOrdenesWithQuote(orderId, quoteId, itemGeos);

      return {
        success: true,
        service: "Split Emails (Graph/Brevo)",
        message: "RFQ split emails sent: deficit to supplier, all items to users.",
        pdfUrl: pdfUrl,
        emailsSent: 2
      };
    } else {
      return {
        success: false,
        message: "User email dispatch failed: " + (userResult ? userResult.message || userResult.error : "Unknown error")
      };
    }
  }

  // Scenario B: ALL IN STOCK - Send only to users
  if (isScenarioB) {
    console.log("SCENARIO B: All in-stock - sending email to users only");

    const routing = getRequirementRouting(details.site || "", routingCategoriesText, String(details.subArea || ""));
    let userEmails = routing.emails || "";

    // If routing returned fallback, use CC field from form instead
    if (userEmails === "procurement@yourcompany.com" || userEmails.trim() === "") {
      userEmails = String(details.cc || "").trim();
    }

    // If still empty, abort and ask user
    if (!userEmails) {
      return {
        success: false,
        message: "No user emails found from Users sheet. Please add emails in the CC field and try again."
      };
    }

    const personalizedBaseBody = personalizeBodyForUsers_(baseBody);
    const userEmailBody = buildRFQEmailBody_(personalizedBaseBody, [], bodegaItems, {
      showDeficit: false,
      showInStock: true
    });

    const userEmailPackage = {
      type: "RFQ",
      refId: String(details.orderId || "PENDING"),
      to: userEmails,
      cc: "",
      subject: subject + " [Fulfilled from Stock]",
      body: userEmailBody + pdfLinkHtml + (signatureHtml ? `<br><br>${signatureHtml}` : ""),
      supplier: String(details.supplier || "N/A"),
      replyTo: String(details.replyTo || Session.getActiveUser().getEmail()),
      senderEmail: senderEmail,
      fallbackProvider: String(details.fallbackProvider || ""),
      pdfUrl: pdfUrl
    };

    console.log("Dispatching User Email (Scenario B)...");
    const userResult = SmartDispatcher(userEmailPackage);

    if (userResult && userResult.success) {
      persistQuote_();
      updateQuoteSheetStatus_(quoteId, "Pending");

      const itemGeos = details.itemGeos || quoteData.items.map(function (item) {
        return item.geo;
      });

      updateQuotedStatus(orderId, quoteId, itemGeos);
      updateOrdenesWithQuote(orderId, quoteId, itemGeos);

      return {
        success: true,
        service: "Graph/Brevo",
        message: "All items fulfilled from stock. Email sent to users only.",
        pdfUrl: pdfUrl,
        emailsSent: 1
      };
    } else {
      return {
        success: false,
        message: "User email dispatch failed: " + (userResult ? userResult.message || userResult.error : "Unknown error")
      };
    }
  }

  // Scenario C: ALL OUT OF STOCK - Send single email to supplier + users CC
  if (isScenarioC) {
    console.log("SCENARIO C: All out of stock - sending single email to supplier + users");

    const routing = getRequirementRouting(details.site || "", routingCategoriesText, String(details.subArea || ""));
    let userEmails = routing.emails || "";

    // If routing returned fallback, use CC field from form instead
    if (userEmails === "procurement@yourcompany.com" || userEmails.trim() === "") {
      userEmails = String(details.cc || "").trim();
    }

    // If no emails available for CC, it's OK in Scenario C - just send to supplier
    if (!userEmails) {
      console.warn("No user emails found for CC - will send only to supplier");
      userEmails = "";
    }

    const emailBody = buildRFQEmailBody_(baseBody, deficitItems, [], {
      showDeficit: true,
      showInStock: false
    });

    const emailPackage = {
      type: "RFQ",
      refId: String(details.orderId || "PENDING"),
      to: String(details.to || ""),
      cc: userEmails,
      subject: subject,
      body: emailBody + (signatureHtml ? `<br><br>${signatureHtml}` : ""),
      supplier: String(details.supplier || "N/A"),
      replyTo: String(details.replyTo || Session.getActiveUser().getEmail()),
      senderEmail: senderEmail,
      fallbackProvider: String(details.fallbackProvider || ""),
      pdfUrl: "" // No PDF link in supplier email (internal PDF still generated + saved)
    };

    console.log("Dispatching Email (Scenario C)...");
    const result = SmartDispatcher(emailPackage);

    if (result && result.success) {
      persistQuote_();
      updateQuoteSheetStatus_(quoteId, "Pending");

      const itemGeos = details.itemGeos || quoteData.items.map(function (item) {
        return item.geo;
      });

      updateQuotedStatus(orderId, quoteId, itemGeos);
      updateOrdenesWithQuote(orderId, quoteId, itemGeos);

      return {
        success: true,
        service: result.service || "Graph/Brevo",
        message: "RFQ email sent to supplier with users CC (all items out of stock).",
        pdfUrl: pdfUrl,
        emailsSent: 1
      };
    } else {
      return {
        success: false,
        message: result.message || result.error || "Unknown Dispatch Error"
      };
    }
  }

  // Fallback: should not reach here, but handle edge case
  return {
    success: false,
    message: "Unable to determine email scenario. Check item stock classification."
  };

  } catch (e) {
    console.error("sendFinalRFQ Error: " + e.toString());
    return {
      success: false,
      message: "Backend Crash: " + e.toString(),
    };
  }
}

  };

  const normalizedSite = String(site || "").trim().toUpperCase();

  for (let i = 1; i < data.length; i++) {
    const itemId = String(data[i][0] || "").trim();
    if (!itemId) continue;

    const locType = String(data[i][2] || "").trim();
    const locationId = String(data[i][3] || "").trim().toUpperCase();
    const available = Number(data[i][6] || 0);

    // Site/Sector stock
    if (normalizedSite && locationId === normalizedSite) {
      result.siteStock[itemId] =
        (result.siteStock[itemId] || 0) + available;
    }else if(locationId.toLowerCase()==="bodega"){
      result.bodegaStock[itemId] =
        (result.bodegaStock[itemId] || 0) + available;

    }

   
  }

  return result;
}

function getReqQuoteTimeline(orderId) {
  const oid = String(orderId || "").trim();
  const empty = { orderId: oid, reqCreated: "", reqCreatedIso: "", reqApprovedBy: "",
    reqApprovedDate: "", reqLastModifiedBy: "", reqLastModified: "",
    reqLastModifiedIso: "", reqModCount: 0, quotes: [] };
  try {
    const all = getQuotesHistory().filter(function (r) {
      return String((r && r.orderId) || "").trim() === oid;
    });
    if (!all.length) return empty;
    const cards = {};
    all.forEach(function (r) {
      const key = [String(r.orderId || "").trim(), String(r.supplier || "").trim(),
        String(r.category || "").trim().toUpperCase()].join("|");
      if (!cards[key]) cards[key] = { supplier: r.supplier || "", quoteIds: [], rows: [] };
      const c = cards[key];
      const qid = String(r.quoteId || "").trim();
      if (qid && c.quoteIds.indexOf(qid) === -1) c.quoteIds.push(qid);
      c.rows.push(r);
    });
    let reqCreated = "", reqCreatedIso = "", reqMod = "", reqModIso = "", reqModBy = "", reqModCount = 0;
    all.forEach(function (r) {
      const cIso = String(r.reqDateIso || "").trim(), cRaw = String(r.reqDate || "").trim();
      if (cIso && (!reqCreatedIso || cIso < reqCreatedIso)) { reqCreatedIso = cIso; reqCreated = cRaw; }
      else if (!reqCreated && !reqCreatedIso && cRaw) { reqCreated = cRaw; }
      const mIso = String(r.reqLastModifiedIso || "").trim(), mRaw = String(r.reqLastModified || "").trim();
      if ((mRaw || mIso) && (!reqMod || (mIso && (!reqModIso || mIso > reqModIso)))) { reqMod = mRaw; reqModIso = mIso; }
      if (!reqModBy && r.reqLastModifiedBy) reqModBy = String(r.reqLastModifiedBy);
      if ((Number(r.reqModCount) || 0) > reqModCount) reqModCount = Number(r.reqModCount) || 0;
    });
    const quotes = Object.keys(cards).map(function (k) {
      const c = cards[k];
      const ev = resolveQuoteTimelineEvents_(c.quoteIds, oid, c.supplier);
      // Card audit: earliest created + newest modified across its rows.
      let cd = "", ci = "", cb = "", cbn = "", md = "", mi = "", mb = "", mbn = "", mc = 0;
      c.rows.forEach(function (r) {
        const cI = String(r.createdIso || "").trim();
        if ((r.createdDate || cI) && (!cd || (cI && (!ci || cI < ci)))) {
          cd = String(r.createdDate || "").trim(); ci = cI;
          cb = String(r.createdBy || "").trim(); cbn = String(r.createdByName || r.createdBy || "").trim();
        }
        const mI = String(r.lastModifiedIso || "").trim();
        if ((r.lastModifiedDate || mI) && (!md || (mI && (!mi || mI > mi)))) {
          md = String(r.lastModifiedDate || "").trim(); mi = mI;
          mb = String(r.lastModifiedBy || "").trim(); mbn = String(r.lastModifiedByName || r.lastModifiedBy || "").trim();
        }
        if ((Number(r.modCount) || 0) > mc) mc = Number(r.modCount) || 0;
      });
      return { supplier: c.supplier, quoteId: c.quoteIds.length ? c.quoteIds[c.quoteIds.length - 1] : "",
        createdDate: cd, createdBy: cbn || cb, lastModifiedDate: md,
        lastModifiedBy: mbn || mb, modCount: mc,
        uploadedDate: ev.uploadedDate, uploadedBy: ev.uploadedByName || ev.uploadedBy,
        approvedBy: ev.approvedByName || ev.approvedBy, approvedDate: ev.approvedDate };
    });
    empty.reqCreated = reqCreated; empty.reqCreatedIso = reqCreatedIso;
    empty.reqLastModifiedBy = reqModBy; empty.reqLastModified = reqMod;
    empty.reqLastModifiedIso = reqModIso; empty.reqModCount = reqModCount;
    empty.quotes = quotes;
    return empty;
  } catch (e) { console.warn("getReqQuoteTimeline failed: " + e.toString()); return empty; }
}

===== getRequirementRouting (Código.js) =====
function getRequirementRouting(siteName, categoryName, subAreaName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  const sName = String(siteName || "").trim();

  // OPTIONAL SubArea routing override (Validation sheet Col V = SubArea,
  // Col W = Emails SubArea). When a SubArea is provided (e.g. for the
  // "Workshop" site) its emails REPLACE the category-based recipients
  // computed from the Users sheet below. Falls back to category routing if
  // the SubArea is unknown or has no emails configured.
  const subAreaRequested = String(subAreaName || "").trim();
  if (subAreaRequested) {
    const subAreaRouting = resolveSubAreaRouting_(sName, subAreaRequested);
    if (subAreaRouting) {
      return subAreaRouting;
    }
    console.warn(
      '[getRequirementRouting] SubArea "' + subAreaRequested + '" for site "' + sName +
      '" not found or has no emails; falling back to category routing.'
    );
  }

  // 1. Suffix Lookup (still uses Site to identify the project code)
  const valSheet = ss.getSheetByName("Validation");
  let suffix = "REQ";
  if (valSheet) {
    const valData = valSheet.getDataRange().getValues();
    for (let i = 1; i < valData.length; i++) {
      if (normalizeUserMatch_(valData[i][0]) === normalizeUserMatch_(sName)) {
        suffix = valData[i][2] || "REQ";
        break;
      }
    }
  }

  // 2. Email Lookup (Users sheet) - header aware so we never read the wrong
  //    column. Canonical layout (AdminVal):
  //    ID, PIN, Name, Email, Position, Site, Role, Active
  const userSheet = ss.getSheetByName("Users");
  let emails = [];

  if (userSheet) {
    const userData = userSheet.getDataRange().getValues();
    const headersUser = userData.length > 0
      ? userData[0].map(function (h) { return normalizeUserMatch_(h); })
      : [];

    const findUserCol = function (names, fallbackIdx) {
      const wanted = Array.isArray(names) ? names : [names];
      for (let ix = 0; ix < wanted.length; ix++) {
        const hit = headersUser.indexOf(wanted[ix]);
        if (hit !== -1) return hit;
      }
      return fallbackIdx;
    };

    // 🔒 FIX: "Position" is the user's category — never assume a fixed column.
    const colEmail     = findUserCol(["email"], 3);
    const colPosition  = findUserCol(["position", "category"], 4);
    const colRole      = findUserCol(["role"], 6);
    const colActive    = findUserCol(["active", "activo"], 7);

    // Accepted categories: may be empty (General), a single category, or a
    // comma/semicolon-separated list collected from the actual quote items.
    const cTokens = String(categoryName || "")
      .split(/[,;]+/)
      .map(function (t) { return normalizeUserMatch_(t); })
      .filter(Boolean);

    const GENERAL_CATEGORIES = [
      "general", "generica", "generico", "sin categoria",
      "sin cat", "uncategorized", "varios",
    ];
    const isGeneralRequest = cTokens.length === 0 ||
      cTokens.every(function (t) { return GENERAL_CATEGORIES.indexOf(t) !== -1; });
    // In General mode only explicit General-ish categories are accepted; in a
    // specific-category RFQ the user Position must match one of the accepted cats.
    const acceptedCats = isGeneralRequest ? GENERAL_CATEGORIES : cTokens;

    for (let j = 1; j < userData.length; j++) {
      const uEmail   = String(userData[j][colEmail] || "").trim();
      const uCat     = normalizeUserMatch_(userData[j][colPosition]);
      const uRole    = normalizeUserMatch_(userData[j][colRole]);
      const uActive  = normalizeUserMatch_(userData[j][colActive]);

      // 🔒 FIX: users without a Position (category) are never added to the CC list.
      if (!uCat) continue;

      const categoryMatches = acceptedCats.indexOf(uCat) !== -1;
      const roleMatches =
        uRole === "office" || uRole === "admin" || uRole === "manager";
      const activeMatches =
        uActive === "1" || uActive === "true" || uActive === "yes";

      if (categoryMatches && roleMatches && activeMatches) {
        if (uEmail && emails.indexOf(uEmail) === -1) {
          emails.push(uEmail);
        }
      }
    }
  }

  return {
    emails: emails.length > 0 ? emails.join(", ") : "procurement@yourcompany.com",
    suffix: suffix,
    matchedUsers: emails,
  };
}

function searchGraphDirectory(term) {
  if (!term || String(term).trim().length < 2) return [];
  const t = String(term).trim();

  try {
    const creds = getAzureGraphCredentials_();
    const accessToken = getMicrosoftGraphToken(creds);

    // Search by displayName startswith
    const nameEndpoint =
      'https://graph.microsoft.com/v1.0/users?$filter=startswith(displayName,\'' + t.replace(/'/g, "\\'") + "')" +
      '&$select=displayName,mail,mailNickname&$top=10&$orderby=displayName';

    const nameOptions = {
      method: 'get',
      headers: {
        'Authorization': 'Bearer ' + accessToken,
        'Accept': 'application/json'
      },
      muteHttpExceptions: true
    };

    const nameResp = UrlFetchApp.fetch(nameEndpoint, nameOptions);
    const nameCode = nameResp.getResponseCode();
    const nameData = JSON.parse(nameResp.getContentText());

    const results = [];

    if (nameCode === 200 && nameData.value) {
      nameData.value.forEach(function(u) {
        const mail = String(u.mail || u.mailNickname || '').toLowerCase().trim();
        if (mail && isCompanyEmail(mail)) {
          results.push({
            displayName: String(u.displayName || '').trim(),
            email: mail
          });
        }
      });
    }

    // Also try exact mail match
    if (t.indexOf('@') !== -1 && results.length === 0) {
      const mailEndpoint =
        'https://graph.microsoft.com/v1.0/users?$filter=mail eq \'' + t.replace(/'/g, "\\'") + "'" +
        '&$select=displayName,mail&$top=5';

      const mailOptions = {
        method: 'get',
        headers: {
          'Authorization': 'Bearer ' + accessToken,
          'Accept': 'application/json'
        },
        muteHttpExceptions: true
      };

      const mailResp = UrlFetchApp.fetch(mailEndpoint, mailOptions);
      const mailCode = mailResp.getResponseCode();
      const mailData = JSON.parse(mailResp.getContentText());

      if (mailCode === 200 && mailData.value) {
        mailData.value.forEach(function(u) {
          const mailAddr = String(u.mail || '').toLowerCase().trim();
          if (mailAddr && isCompanyEmail(mailAddr)) {
            results.push({
              displayName: String(u.displayName || '').trim(),
              email: mailAddr
            });
          }
        });
      }
    }

    return results.slice(0, 10);
  } catch (e) {
    console.warn('Graph directory search failed:', e.toString());
    return [];
  }
}

function getEmailSuggestions(searchTerm) {
  if (!searchTerm || String(searchTerm).trim().length === 0) return [];

  const term = String(searchTerm).trim().toLowerCase();
  const results = [];
  const seen = {};

  const addResult = function(displayName, email, source) {
    if (!email || !displayName) return;
    const key = email.toLowerCase();
    if (Object.prototype.hasOwnProperty.call(seen, key)) return;
    seen[key] = true;
    results.push({
      label: displayName + ' <' + email + '>',
      value: email,
      source: source
    });
  };

  // 1. People sheet (company domain emails only)
  try {
    const people = getPeople() || [];
    people.forEach(function(p) {
      if (!p || !p.email) return;
      const email = String(p.email).toLowerCase().trim();
      if (!isCompanyEmail(email)) return;
      const name = String(p.name || '').trim();
      if (!name) return;
      if (name.toLowerCase().indexOf(term) !== -1 || email.indexOf(term) !== -1) {
        addResult(name, email, 'people');
      }
    });
  } catch (e) {
    console.warn('People sheet suggestion lookup failed:', e.toString());
  }

  // 2. Suppliers (from Validation sheet directly)
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const valSheet = ss.getSheetByName('Validation');
    let suppliers = [];
    if (valSheet) {
      const valData = valSheet.getDataRange().getValues();
      for (let i = 1; i < valData.length; i++) {
        if (valData[i][15]) {
          suppliers.push({ name: String(valData[i][15] || '').trim(), email: String(valData[i][16] || '').trim() });
        }
      }
    }
    suppliers.forEach(function(s) {
      if (!s || !s.email) return;
      const email = String(s.email).toLowerCase().trim();
      if (!email) return;
      const name = String(s.name || '').trim();
      if (!name) return;
      if (name.toLowerCase().indexOf(term) !== -1 || email.indexOf(term) !== -1) {
        addResult(name, email, 'supplier');
      }
    });
  } catch (e) {
    console.warn('Supplier suggestion lookup failed:', e.toString());
  }

  // 3. Microsoft Graph directory (company emails only)
  try {
    const graphResults = searchGraphDirectory(searchTerm);
    graphResults.forEach(function(u) {
      addResult(u.displayName || u.email, u.email, 'graph');
    });
  } catch (e) {
    console.warn('Graph suggestion lookup failed:', e.toString());
  }

  // Sort: Graph first (canonical names), then People, then Suppliers
  const sourceOrder = { graph: 1, people: 2, supplier: 3 };
  results.sort(function(a, b) {
    const sa = sourceOrder[a.source] || 99;
    const sb = sourceOrder[b.source] || 99;
    if (sa !== sb) return sa - sb;
    return (a.label || '').localeCompare(b.label || '');
  });

  return results.slice(0, 10);
}

function getRequirementsHistory() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Ordenes");

    if (!sheet) return [];

    const data = sheet.getDataRange().getValues();

    if (data.length <= 1) return [];

    // Tag each row with its physical sheet row number (1-based; row 1 is the
    // header) BEFORE filtering, so the edit modal can pair each item back to
    // the exact row on save. The history is returned newest-first below, so
    // the server must NOT pair items to rows by array index (it used to,
    // which wrote each item's Geo/P/N/Desc onto the wrong row).
    const cleanData = data
      .slice(1)
      .map((row, idx) => ({ row: row, rowNum: idx + 2 }))
      .filter((entry) => entry.row[0] !== "" && entry.row[0] !== null)
      .filter((entry) => {
        // Exclude drafts from history view
        const estado = String(entry.row[7] || "").trim();
        return estado !== "Draft" && !String(entry.row[0] || "").trim().startsWith("DRAFT-");
      });

    // Header-aware lookup of the optional "SubArea" / "Emails" / "SubArea No"
    // and "Notes" columns, plus the requirement AUDIT columns
    // ("lastModifiedBy" / "lastModifiedDate" / "modCount") written by CreateReqGS
    // at creation and by every requirement edit since. The audit headers are
    // matched case-insensitively AND with the spaced spelling ("Last Modified By"),
    // because the sheet has existed in both forms; a -1 index simply means the
    // column is missing (legacy sheet / never edited) and the consumer shows "-".
    const headers = data[0];
    let subAreaIdx = -1;
    let emailsIdx = -1;
    let subAreaNoIdx = -1;
    let notesIdx = -1;
    let itemNotesIdx = -1;
    let supplierNotesIdx = -1;
    let itemImagesIdx = -1;
    let lastModByIdx = -1;
    let lastModDateIdx = -1;
    let modCountIdx = -1;
    for (let i = 0; i < headers.length; i++) {
      const headerName = String(headers[i] || "").trim();
      const headerKey = headerName.toLowerCase();
      if (headerName === "SubArea") subAreaIdx = i;
      if (headerName === "Emails") emailsIdx = i;
      if (headerName === "SubArea No") subAreaNoIdx = i;
      if (headerName === "Notes") notesIdx = i;
      if (headerKey === "item notes") itemNotesIdx = i;
      if (headerKey === "supplier notes") supplierNotesIdx = i;
      if (headerKey === "item images") itemImagesIdx = i;
      if (headerKey === "lastmodifiedby" || headerKey === "last modified by") lastModByIdx = i;
      if (headerKey === "lastmodifieddate" || headerKey === "last modified date") lastModDateIdx = i;
      if (headerKey === "modcount" || headerKey === "mod count") modCountIdx = i;
    }

    // Requirement audit date cells arrive either as a real Date (CreateReqGS
    // stamps new Date()) or as the "yyyy-MM-dd HH:mm:ss" text ViewEditReqGS
    // writes. Both are normalized to { text, iso }: `text` is the display value
    // (dd-MM-yyyy HH:mm in the script timezone, or the original string when it
    // cannot be parsed) and `iso` is a sortable timestamp for the client.
    const auditDate_ = function (v) {
      if (Object.prototype.toString.call(v) === "[object Date]") {
        if (isNaN(v.getTime())) return { text: "", iso: "" };
        return {
          text: Utilities.formatDate(v, Session.getScriptTimeZone(), "dd-MM-yyyy HH:mm"),
          iso: v.toISOString(),
        };
      }
      const raw = String(v == null ? "" : v).trim();
      if (!raw) return { text: "", iso: "" };
      const parsed = new Date(raw);
      if (isNaN(parsed.getTime())) return { text: raw, iso: "" };
      return {
        text: Utilities.formatDate(parsed, Session.getScriptTimeZone(), "dd-MM-yyyy HH:mm"),
        iso: parsed.toISOString(),
      };
    };

    return cleanData
      .map((entry) => {
        const row = entry.row;
        // Machine-readable requirement date next to the locale string: consumers
        // that must sort or format the Req date (e.g. the Quote Mgmt requirement
        // header) cannot rely on toLocaleString(), which is locale dependent.
        const reqDate = parseRequirementDate_(row[1]);
        const reqDateValid = !!reqDate;
        // Single parse of the requirement audit date (text + iso computed once).
        const reqLastMod = lastModDateIdx >= 0
          ? auditDate_(row[lastModDateIdx])
          : { text: "", iso: "" };
        return {
          rowNum: entry.rowNum,
          order: row[0] || "N/A",
          fecha: row[1] ? row[1].toLocaleString() : "",
          fechaIso: reqDateValid ? reqDate.toISOString() : "",
          pn: row[2] || "",
          geo: row[3] || "",
          faena: row[4] || "",
          desc: row[5] || "",
          cant: row[6] || 0,
          estado: row[7] || "Pending",
          user: row[8],
          category: row[12] || "General",
          pdfUrl: row[13] || "",
          subArea: subAreaIdx >= 0 ? (row[subAreaIdx] || "") : "",
          emails: emailsIdx >= 0 ? (row[emailsIdx] || "") : "",
          subAreaNo: subAreaNoIdx >= 0 ? (row[subAreaNoIdx] || "") : "",
          notes: notesIdx >= 0 ? (row[notesIdx] || "") : "",
          // Per-item notes (unique to this requirement row: Order + Geo)
          itemNotes: itemNotesIdx >= 0 ? (row[itemNotesIdx] || "") : "",
          // Supplier-facing notes ("Supplier Notes" column): printed in the
          // supplier RFQ (quote request) email only.
          supplierNotes: supplierNotesIdx >= 0 ? (row[supplierNotesIdx] || "") : "",
          // Per-item photos ("Item Images" column): comma-separated Drive URLs
          // uploaded from the Create Req cart. Returned as the raw cell string;
          // consumers normalize with reqItemImageUrls_ / normalizeCartImages.
          images: itemImagesIdx >= 0 ? (row[itemImagesIdx] || "") : "",
          // Requirement AUDIT trail (last requirement edit). Joined onto the
          // Quote Mgmt requirement header's info ("i") timeline: an untouched
          // requirement reports its creation values (CreateReqGS stamps
          // lastModifiedDate at creation; lastModifiedBy stays blank until the
          // first edit) and a legacy sheet without these columns reports "-".
          reqLastModifiedBy: lastModByIdx >= 0 ? String(row[lastModByIdx] || "").trim() : "",
          reqLastModified: reqLastMod.text,
          reqLastModifiedIso: reqLastMod.iso,
          reqModCount: modCountIdx >= 0 ? (Number(row[modCountIdx]) || 0) : 0,
        };
      }

function getScannerData(startDateStr, endDateStr) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const countSheet = getOrCreateSheet(ss, "StockCounts", ["CountID", "Date", "UserName", "Status", "Location"]);
    const lastRow = countSheet.getLastRow();
    
    let history = [];
    
    if (lastRow > 1) {
      const numRows = lastRow - 1;
      // Fetch ALL rows (not just last 15) so we can sort by date properly
      const rawData = countSheet.getRange(2, 1, numRows, 5).getValues();
      
      // Map and convert Date objects to ISO strings for safe transmission
      history = rawData.map(row => {
        return [
          row[0], // ID
          row[1] instanceof Date ? row[1].toISOString() : row[1], // Convert Date to ISO String
          row[2], // User
          row[3], // Status
          row[4]  // Location
        ];
      });

      // Sort by date descending (newest first) so the most recent counts appear at the top
      history.sort((a, b) => {
        const dateA = a[1] instanceof Date ? a[1] : new Date(a[1]);
        const dateB = b[1] instanceof Date ? b[1] : new Date(b[1]);
        return dateB - dateA; // Descending: newest first
      });

      // Apply date range filter if provided (inclusive on both ends)
      if (startDateStr || endDateStr) {
        const fromDate = startDateStr ? new Date(startDateStr + "T00:00:00") : null;
        const toDate = endDateStr ? new Date(endDateStr + "T23:59:59") : null;
        history = history.filter(row => {
          const rowDate = row[1] instanceof Date ? row[1] : new Date(row[1]);
          if (!rowDate || isNaN(rowDate.getTime())) return false;
          if (fromDate && rowDate < fromDate) return false;
          if (toDate && rowDate > toDate) return false;
          return true;
        });
      }

      // Take the most recent 50 entries (increased from 15 to show more history)
      history = history.slice(0, 50);
    }

    const result = { history: history };
    console.log("Returning to frontend:", JSON.stringify(result));
    return result;

  } catch (e) {
    console.error("Scanner Backend Error: " + e.toString());
    return { history: [] };
  }
}

        name: data[i][18],
        symbol: data[i][19] || "",
      });
    }

    // Col V (21): SubArea, Col W (22): Emails SubArea, Col X (23): Suffix.
    // SubAreas belong to the "Workshop" site (the only site that has them),
    // regardless of what may appear in Col A on those rows.
    if (data[i][21]) {
      const subAreaName = String(data[i][21]).trim();
      const subAreaKey = normalizeUserMatch_("workshop") + "|" + normalizeUserMatch_(subAreaName);

        subAreaKeys.add(subAreaKey);
        subAreas.push({
          name: subAreaName,
          email: String(data[i][22] || "").trim(),
          site: "Workshop",
          suffix: String(data[i][23] || "").trim(),
        });
      }
    }
  }

  return {
    categories: categories,
    subCategories: subCategories,
    brands: brands,
    suppliers: suppliers,
    sites: sites,
    currencies: currencies,
    subAreas: subAreas,
  };

      if (oRigId === rigId) {
        details.operations.push({
          date: Utilities.formatDate(new Date(date), Session.getScriptTimeZone(), "yyyy-MM-dd"),
          description: description
        });
      }
    }
  }

  // 5. Query Safety Metrics and Camera Statuses
  const safetySheet = ss.getSheetByName("Op_Safety_Cameras");
  if (safetySheet) {
    const safetyData = safetySheet.getDataRange().getValues();
    for (let i = 1; i < safetyData.length; i++) {
      const [sRigId, score, incidents, lastIncident, cam1, cam2, cam3] = safetyData[i];
      if (sRigId === rigId) {
        details.safety = {

function processAndSaveBulkPO(csvData, selectedCategories, cachedInventory, fileExportOption, columnIndexes, userDateFormat) {
  try {

    var errors = [];

    var iIdx = columnIndexes.item;
    var dIdx = columnIndexes.desc;
    var qIdx = columnIndexes.qty;
    var cIdx = columnIndexes.cost;

    // --- 1. DATA PARSING AND SANITIZATION LOOP ---
    for (var i = 1; i < csvData.length; i++) {
      var row = csvData[i];
      if (!row || row.length <= Math.max(dteIdx, invIdx, vIdx, iIdx, qIdx, cIdx) || !row[invIdx]) continue;

      }
      
      var cleanGPDate = formatToDDMMYYYY(rawDate, userDateFormat);
      var baseCost = cost.toString().replace(/[\$\s,]/g, '');
      var integerCost = baseCost.split('.')[0]; 
      
      if (!poGroups[invNum]) {
        // Unpack composite payload properties safely
        var mappedConfig = selectedCategories[invNum] || {};
        var assignedCategory = (typeof mappedConfig === 'object') ? (mappedConfig.category || "OPS") : "OPS";

          docDate: cleanGPDate,

        };
      }

    }
    
    if (Object.keys(poGroups).length === 0) {
      return { success: false, msg: "No valid transaction data rows compiled successfully.", errors: errors };
    }

    var macroContentText = buildMacroContent(poGroups);
    var csvContentText   = buildFlatCsvContent(poGroups); // <-- Places currency at the end of the line

    var archiveName = baseFileName + ".zip";

function processPendingHSEPDFs() {
  const hseSheet = getOrCreateHSESignaturesSheet();
  const data = hseSheet.getDataRange().getValues();

  if (data.length < 2) {
    console.log("No HSE signature rows to process.");
    return { success: true, processed: 0 };
  }

  const headers = data[0];
  const colMap = {};
  headers.forEach((h, i) => { colMap[String(h).trim()] = i; });

  const idxMovement = colMap["MovementID"];
  const idxWorkerId = colMap["WorkerID"];
  const idxWorkerName = colMap["WorkerName"];
  const idxFingerprint = colMap["FingerprintBase64"];
  const idxScore = colMap["MatchScore"];
  const idxTimestamp = colMap["MatchTimestamp"];
  const idxItemsCount = colMap["ItemsCount"];
  const idxPdfLink = colMap["PDFDriveLink"];
  const idxStatus = colMap["Status"] !== undefined ? colMap["Status"] : 8;
  const idxItemsJson = colMap["HSEItemsJSON"] !== undefined ? colMap["HSEItemsJSON"] : 9;

  let processed = 0;

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const status = String(row[idxStatus] || "").trim().toUpperCase();

    if (status !== "PENDING") continue;

    const movementId = String(row[idxMovement] || "");
    const workerRut = String(row[idxWorkerId] || "");
    const workerName = String(row[idxWorkerName] || "");
    const fingerprintBase64 = String(row[idxFingerprint] || "");
    const matchScore = Number(row[idxScore]) || 0;
    const timestamp = row[idxTimestamp] instanceof Date ? row[idxTimestamp] : new Date(row[idxTimestamp]);

    let hseItems = [];
    try {
      const rawItems = String(row[idxItemsJson] || "").trim();
      if (rawItems) {
        hseItems = JSON.parse(rawItems);
      }
    } catch (e) {
      console.error("Failed to parse HSEItemsJSON for movement " + movementId + ": " + e.toString());
      hseItems = [];
    }

    try {
      // Mark as processing to avoid duplicate attempts on concurrent runs
      hseSheet.getRange(i + 1, idxStatus + 1).setValue("PROCESSING");

      const pdfData = {
        movementId: movementId,
        timestamp: timestamp,
        workerName: workerName,
        workerRut: workerRut,
        fingerprintBase64: fingerprintBase64,
        matchScore: matchScore,
        hseItems: hseItems
      };

      const pdfBlob = generateHSEReceiptPDF(pdfData);
      const driveLink = saveHSEPDFToDrive(movementId, workerRut, workerName, pdfBlob);

      // Update the row with the Drive link and mark as DONE
      hseSheet.getRange(i + 1, idxPdfLink + 1).setValue(driveLink);
      hseSheet.getRange(i + 1, idxStatus + 1).setValue("DONE");

      processed++;
      console.log("HSE PDF generated in background for movement: " + movementId);
    } catch (e) {
      console.error("HSE PDF background processing failed for movement " + movementId + ": " + e.toString());
      // Reset to PENDING so it can be retried on the next run
      hseSheet.getRange(i + 1, idxStatus + 1).setValue("PENDING");
    }
  }

  console.log("processPendingHSEPDFs finished. Processed: " + processed);
  return { success: true, processed: processed };
}

  if (!nameValue) {
    throw new Error('El nombre o identificador no puede estar vacío.');
  }

  if (isEdit && assetData._rowIndex) {
    // Edit existing cell
    sheet.getRange(assetData._rowIndex, colIndex).setValue(nameValue);
  } else {
    // Add new asset: Find first empty row in this specific column or append
    const lastRow = sheet.getLastRow();
    let targetRow = 0;

      const colValues = sheet.getRange(2, colIndex, lastRow - 1, 1).getValues();

        if (String(colValues[i][0] || '').trim() === '') {
          targetRow = i + 2;
          break;
        }
      }
    }

    if (targetRow === 0) {
      targetRow = Math.max(lastRow + 1, 2);
    }

function saveBundle(payload) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {

    const ss = SpreadsheetApp.getActiveSpreadsheet();

    const bundleSheet = ss.getSheetByName("Bundles");
    const itemSheet = ss.getSheetByName("Bundle_Items");

    const bundleData = bundleSheet.getDataRange().getValues();
    const headers = bundleData[0];

    const colId = headers.indexOf("Bundle ID");
    const colName = headers.indexOf("Bundle Name");
    const colActive = headers.indexOf("Active");

    const bundleId = payload.bundleId;
    const name = payload.name;
    const isActive = payload.isActive;
    const items = payload.items || [];

    // check if exists
    const existingIndex = bundleData.slice(1)
      .findIndex(r => String(r[colId]) === String(bundleId));

    if (existingIndex === -1) {
      // CREATE
      bundleSheet.appendRow([bundleId, name, isActive]);
    } else {
      // UPDATE
      const row = existingIndex + 2;
      bundleSheet.getRange(row, colName + 1).setValue(name);
      bundleSheet.getRange(row, colActive + 1).setValue(isActive);
    }

    // DELETE OLD ITEMS
    const itemData = itemSheet.getDataRange().getValues();
    const iHeaders = itemData[0];

    const iBundleCol = iHeaders.indexOf("Bundle ID");

    const remaining = itemData.filter((r, idx) => {
      if (idx === 0) return true;
      return String(r[iBundleCol]) !== String(bundleId);
    });

    itemSheet.clearContents();
    itemSheet.getRange(1, 1, remaining.length, remaining[0].length)
      .setValues(remaining);

    // ADD NEW ITEMS
    const newRows = items.map(i => [
      bundleId,
      i.geo,
      i.qty
    ]);

    if (newRows.length > 0) {
      itemSheet.getRange(

function saveDraft(shipments) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Ordenes");
  const timestamp = new Date();

  try {
    const draftId = "DRAFT-" + timestamp.getTime();

    shipments.forEach((shipment) => {
      // Column indices are resolved once per shipment (the helper creates the
      // "Notes" column on first use) and reused for every draft row.
      const headerRow = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 18)).getValues()[0];
      const headerNames = headerRow.map(function (h) { return String(h || "").trim(); });
      const subAreaColIdx = headerNames.indexOf("SubArea");
      const emailsColIdx = headerNames.indexOf("Emails");
      const subAreaNoColIdx = headerNames.indexOf("SubArea No");
      // "Notes" keeps the draft's "Additional Message (Body)" so a draft
      // re-opened (and later the View/Edit Requirement screen) still shows it.
      // "Item Notes" / "Supplier Notes" keep both per-item notes the same way.
      const notesColIdx = getOrdenesNotesColumn_(sheet);
      const itemNotesColIdx = getOrdenesItemNotesColumn_(sheet);
      const supplierNotesColIdx = getOrdenesSupplierNotesColumn_(sheet);
      // Photos attached to the line in the cart. Stored as ONE comma-separated
      // cell per row so a draft keeps its attachments when re-opened.
      const itemImagesColIdx = getOrdenesItemImagesColumn_(sheet);
      const notesToSave = cleanReqNotes_(shipment.body);

      const rowsToSave = shipment.items.map((item) => {
        const row = [
          draftId, timestamp, item.pn, item.geo, shipment.site,
          item.es || item.en, item.qty, "Draft", shipment.userName,
          "", "", "", shipment.category, "", "", "", new Date(), 1
        ];

        const targetWidth = Math.max(18, subAreaColIdx + 1, emailsColIdx + 1, subAreaNoColIdx + 1, notesColIdx + 1, itemNotesColIdx + 1, supplierNotesColIdx + 1, itemImagesColIdx + 1);
        while (row.length < targetWidth) row.push("");

        if (subAreaColIdx >= 0) row[subAreaColIdx] = shipment.subArea || "";
        if (emailsColIdx >= 0) row[emailsColIdx] = shipment.to || "";
        if (subAreaNoColIdx >= 0) row[subAreaNoColIdx] = shipment.subAreaFolio || "";
        if (notesColIdx >= 0) row[notesColIdx] = notesToSave;
        if (itemNotesColIdx >= 0) row[itemNotesColIdx] = cleanReqNotes_(item && item.itemNotes);
        if (supplierNotesColIdx >= 0) row[supplierNotesColIdx] = cleanReqNotes_(item && item.supplierNotes);
        if (itemImagesColIdx >= 0) row[itemImagesColIdx] = serializeItemImages_(item && item.images);

        return row;
      });

      const saveWidth = Math.max(18, rowsToSave[0].length);
      sheet.getRange(sheet.getLastRow() + 1, 1, rowsToSave.length, saveWidth).setValues(rowsToSave);
    });

    console.log("Draft saved successfully:", draftId);
    return { success: true, draftId: draftId };
  } catch (e) {
    console.error("Error saving draft: " + e.toString());
    return { success: false, message: e.toString() };
  }
}


/**
 * GET DRAFTS - Retrieves all drafts from Ordenes sheet
 * Returns array of draft objects grouped by draft ID
 */
function getDrafts() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Ordenes");

    if (!sheet) return [];

    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) return [];

    const headers = data[0];
    const colMap = {};
    headers.forEach((h, i) => { colMap[String(h).trim()] = i; });

    const drafts = {};

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      const orderId = String(row[0] || "").trim();
      const estado = String(row[colMap["Estado"] !== undefined ? colMap["Estado"] : 7] || "").trim();

      if (orderId.startsWith("DRAFT-") && estado === "Draft") {
        if (!drafts[orderId]) {
          drafts[orderId] = {
            draftId: orderId,
            fecha: row[1] ? new Date(row[1]).toISOString() : "",
            site: row[colMap["Faena"] !== undefined ? colMap["Faena"] : 4] || "",
            user: row[colMap["User"] !== undefined ? colMap["User"] : 8] || "",
            category: row[colMap["Category"] !== undefined ? colMap["Category"] : 12] || "General",
            subArea: row[colMap["SubArea"]] || "",
            emails: row[colMap["Emails"]] || "",
            subAreaNo: row[colMap["SubArea No"]] || "",
            notes: row[colMap["Notes"]] || "",
            items: []
          };
        }

        drafts[orderId].items.push({
          pn: row[colMap["P/N"] !== undefined ? colMap["P/N"] : 2] || "",
          geo: row[colMap["Geo Code"] !== undefined ? colMap["Geo Code"] : 3] || "",
          desc: row[colMap["Desc Es"] !== undefined ? colMap["Desc Es"] : 5] || "",
          es: row[colMap["Desc Es"] !== undefined ? colMap["Desc Es"] : 5] || "",
          qty: row[colMap["Cantidad"] !== undefined ? colMap["Cantidad"] : 6] || 0,
          itemNotes: row[colMap["Item Notes"]] || "",
          supplierNotes: row[colMap["Supplier Notes"]] || "",
          // Attached photos come back as the raw comma-separated cell; the client
          // splits it back into an array of {url,name} on loadDraft().
          images: row[colMap["Item Images"]] || ""
        });
      }
    }

    return Object.values(drafts).sort((a, b) => new Date(b.fecha) - new Date(a.fecha));
  } catch (e) {
    console.error("Error getting drafts: " + e.toString());
    return [];
  }
}

function deleteDraftRows(draftId) {

// ============================================================
// ITEM IMAGE ATTACHMENTS (Create Req cart)
// ------------------------------------------------------------
// Each cart line can carry one or more photos. They are uploaded to Drive the
// MOMENT the user picks them (not at submit time) so:
//   - the Ordenes "Item Images" cell only ever holds Drive URLs;
//   - the payload sent to processMassOrders stays tiny (a big order with many
//     photos would otherwise blow the google.script.run ~50 MB payload limit);
//   - the user immediately sees whether a photo actually stuck.
//
// Payload: { site, files: [{ name, mimeType, base64 }] }
// Returns: { success: true, urls: [...] } | { success: false, error }
// ============================================================

var REQ_IMAGES_MAX_FILES_PER_CALL = 8;
var REQ_IMAGES_MAX_FILES_PER_ITEM = 8;

/**
 * Resolves (creating as needed) the folder that stores requirement item images:
 * <spreadsheet parent>/Req/<Site>/Images - the same "Req/<Site>" tree the
 * requirement PDFs already live in (see orchestrateRequirementPDF), so photos
 * sit next to the PDF they relate to.
 * @param {string} site Project site name
 * @return {Folder} target folder
 */
function getReqImagesFolder_(site) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ssFile = DriveApp.getFileById(ss.getId());
  var parents = ssFile.getParents();
  var parentFolder = parents.hasNext() ? parents.next() : DriveApp.getRootFolder();

  var reqFolder = shared_getOrCreateFolder(parentFolder, "Req");
  var siteName = String(site || "").trim() || "General";
  var siteFolder = shared_getOrCreateFolder(reqFolder, siteName);
  return shared_getOrCreateFolder(siteFolder, "Images");
}

/**
 * Uploads the images attached to ONE cart line to Drive and returns their URLs.
 * Mirrors uploadQuoteAttachment (QuoteMgmtGS.js): the client downscales/re-encodes
 * to JPEG, the server stores the blob as-is and shares it view-by-link.
 * Each file is stored separately (one URL per photo) - they are NOT repackaged
 * into a PDF, because the requirement screen shows them as a thumbnail strip.
 * @param {{site: string, files: Array<{name: string, mimeType: string, base64: string}>}} payload
 * @return {{success: boolean, urls: Array<string>, error?: string}}
 */
function uploadRequirementItemImages(payload) {
  var created = [];
  try {
    if (!payload || !Array.isArray(payload.files) || payload.files.length === 0) {
      return { success: false, error: "No file content provided." };
    }
    if (payload.files.length > REQ_IMAGES_MAX_FILES_PER_CALL) {
      return { success: false, error: "Too many files in one upload (max " + REQ_IMAGES_MAX_FILES_PER_CALL + ")." };
    }

    var folder = getReqImagesFolder_(payload.site);
    var stamp = new Date().getTime();
    var urls = [];

    for (var i = 0; i < payload.files.length; i++) {
      var f = payload.files[i];
      if (!f || !f.base64) continue;

      var raw = String(f.base64 || "");
      // Tolerate a full data URL being sent by mistake (take the payload half).
      var commaAt = raw.indexOf(",");
      if (raw.indexOf("data:") === 0 && commaAt !== -1) raw = raw.substring(commaAt + 1);
      if (!raw) continue;

      var mime = String(f.mimeType || "image/jpeg").toLowerCase();
      var ext = mime.indexOf("png") !== -1 ? ".png" : (mime.indexOf("webp") !== -1 ? ".webp" : ".jpg");

      // Original name (sans extension) + index + timestamp so re-uploads of the
      // same photo never overwrite each other or a previous requirement's file.
      var baseName = sanitizeFileName_(String(f.name || "image").replace(/\.[^.]+$/, "").trim() || "image");
      var fileName = baseName + "_" + stamp + "_" + (i + 1) + ext;

      var blob = Utilities.newBlob(Utilities.base64Decode(raw), mime, fileName);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      created.push(file);
      urls.push(file.getUrl());
    }

    if (!urls.length) {
      return { success: false, error: "No readable image content was provided." };
    }

    console.log("uploadRequirementItemImages: stored " + urls.length + " image(s) for site " + payload.site);
    return { success: true, urls: urls };
  } catch (e) {
    // Never leave orphaned half-batches in Drive: a failure on file N trashes
    // the files already created so the folder does not accumulate junk.
    created.forEach(function (f) {
      try { f.setTrashed(true); } catch (e2) { /* ignore */ }
    });
    console.warn("uploadRequirementItemImages failed: " + e.toString());
    return { success: false, error: String(e) };
  }
}

function checkAndSendFuelAlert() {
  const balanceData = getFuelBalance();
  const currentBalance = balanceData.balance;
  const THRESHOLD = 200;
  
  const scriptProperties = PropertiesService.getScriptProperties();
  const alertStatus = scriptProperties.getProperty('FUEL_ALERT_SENT');

  // If balance is critical and an alert hasn't been sent yet
  if (currentBalance <= THRESHOLD && alertStatus !== 'true') {
    sendFuelAlertEmail(balanceData);
    // Mark alert as sent to prevent spamming
    scriptProperties.setProperty('FUEL_ALERT_SENT', 'true');
  } else if (currentBalance > THRESHOLD) {
    // Reset the alert flag if the fuel has been refilled above the threshold
    scriptProperties.setProperty('FUEL_ALERT_SENT', 'false');
  }
}

===== saveMinMaxImport (MinMaxGS.js) =====
function saveMinMaxImport(rows, mode) {
  if (!rows || !Array.isArray(rows) || !rows.length) {
    throw new Error("saveMinMaxImport: rows array required");
  }
  const conflictMode = mode === "newOnly" ? "newOnly" : "replace";

  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    // Server-side re-validation: valid rows only
    const valid = [];
    let failed = 0;
    rows.forEach((entry) => {
      const site = String(entry.site || "").trim();
      const geo = String(entry.geo || "").trim();
      if (!site || !geo) { failed++; return; }
      const min = Math.max(0, Number(entry.min) || 0);
      const max = Math.max(0, Number(entry.max) || 0);
      valid.push({ site: site, geo: geo, min: min, max: max });
    });

    // De-duplicate within the import itself (last wins)
    const byKey = new Map();
    valid.forEach((r) => {
      byKey.set(r.site.toUpperCase() + "__" + r.geo.toUpperCase(), r);
    });

    const sheet = getMinMaxSheet_();
    const data = sheet.getDataRange().getValues();
    const existing = new Set();
    for (let i = 1; i < data.length; i++) {
      const site = String(data[i][0] || "").trim().toUpperCase();
      const geo = String(data[i][1] || "").trim().toUpperCase();
      if (site && geo) existing.add(site + "__" + geo);
    }

    // Split according to conflict mode
    const toWrite = [];
    let skipped = 0;
    byKey.forEach((r, key) => {
      if (existing.has(key)) {
        if (conflictMode === "replace") toWrite.push(r);
        else skipped++;
      } else {
        toWrite.push(r);
      }
    });

    let saved = 0;
    if (toWrite.length) {
      const res = saveMinMaxItems(toWrite);
      saved = res.count || 0;
    }

    return { saved: saved, skipped: skipped, failed: failed };
  } finally {
    lock.releaseLock();
  }
}

===== saveMinMaxItems (MinMaxGS.js) =====
function saveMinMaxItems(rows) {
  if (!rows || !Array.isArray(rows)) {
    throw new Error("saveMinMaxItems: rows array required");
  }

  const sheet = getMinMaxSheet_();
  const data = sheet.getDataRange().getValues();
  const now = new Date();
  const user = Session.getActiveUser
    ? String(Session.getActiveUser().getEmail() || "") || "System"
    : "System";

  // Build lookup of existing row indices keyed by `SITE__GEO`.
  const keyToRow = new Map();
  for (let i = 1; i < data.length; i++) {
    const site = String(data[i][0] || "").trim().toUpperCase();
    const geo = String(data[i][1] || "").trim().toUpperCase();
    if (site && geo) keyToRow.set(site + "__" + geo, i + 1);
  }

  let count = 0;
  rows.forEach((entry) => {
    const site = String(entry.site || "").trim();
    const geo = String(entry.geo || "").trim();
    if (!site || !geo) return;

    const min = Math.max(0, Number(entry.min) || 0);
    const max = Math.max(0, Number(entry.max) || 0);
    const key = site.toUpperCase() + "__" + geo.toUpperCase();

    const existingRow = keyToRow.get(key);
    if (existingRow) {
      sheet.getRange(existingRow, 1, 1, 6).setValues([
        [site, geo, min, max, now, user],
      ]);
    } else {
      sheet.appendRow([site, geo, min, max, now, user]);
      keyToRow.set(key, sheet.getLastRow());
    }
    count++;
  });

  return { success: true, count };
}

function sendFinalRFQ(details) {

  try {
    
  const flags = getQuoteConfigFlags_();

  // =====================================================
  // PHASE 2: EXTRACT AND CLASSIFY ITEMS
  // =====================================================
  const deficitItems = details.deficitItems || [];
  const bodegaItems = details.bodegaItems || [];

  });

  // Scenario A: Mixed inventory (both deficit & in-stock)

  const isScenarioB = deficitItems.length === 0 && bodegaItems.length > 0;
  const isScenarioC = deficitItems.length > 0 && bodegaItems.length === 0;

      if (it && it.freight !== undefined && it.freight !== null && String(it.freight).trim() !== "") {
        return normalizeFreight_(it.freight);
      }
      const stored = rowMap[clean(it && it.quoteId) + "|" + clean(it && it.origGeo)];
      if (stored === undefined || colFreight === -1) return 0;
      return normalizeFreight_(rows[stored][colFreight]);
    };
    const freightCarrierItemIndex_ = (function () {
      for (let i = 0; i < payload.items.length; i++) {
        if (effectiveFreightOfItem_(payload.items[i]) > 0) return i;
      }
      return -1;
    })();
    if (freightCarrierItemIndex_ !== -1) {
      console.log("💰 Transport carrier for this quote: payload item #" + freightCarrierItemIndex_);
    }

    // Process and collect diffs but do not regenerate PDF yet
    payload.items.forEach((item, itemIndex) => {
      const key = clean(item.quoteId) + "|" + clean(item.origGeo);

      let rowIndex = rowMap[key];

      // Debug logging to help diagnose lookup issues
      if (rowIndex === undefined) {
        console.log("DEBUG: Row lookup failed for key:", key);
        console.log("DEBUG: Available rowMap keys:", Object.keys(rowMap));
        console.log("DEBUG: Item quoteId:", item.quoteId, "origGeo:", item.origGeo);

        // Fallback: Try to find row by iterating through all rows
        // This handles cases where there might be whitespace or case differences
        for (let i = 0; i < rows.length; i++) {
          const rowQuoteId = clean(rows[i][colQuoteId]);
          const rowGeo = clean(rows[i][colGeo]);
          if (rowQuoteId === clean(item.quoteId) && rowGeo === clean(item.origGeo)) {
            console.log("DEBUG: Found row by fallback search at index:", i);
            rowIndex = i;
            break;
          }
        }

        if (rowIndex === undefined) {
          // Additional fallback: Try matching by Geo Code only (for same Quote ID group)
          // This handles cases where Geo Code might have been modified
          for (let i = 0; i < rows.length; i++) {
            const rowQuoteId = clean(rows[i][colQuoteId]);
            if (rowQuoteId === clean(item.quoteId)) {
              console.log("DEBUG: Found row by Quote ID only at index:", i, "row geo:", rows[i][colGeo]);
              rowIndex = i;
              break;
            }
          }
        }

        if (rowIndex === undefined) {
          throw new Error("No matching row found for: " + JSON.stringify({ quoteId: item.quoteId, geo: item.origGeo, key: key }));
        }
      }

      const row = rows[rowIndex];

      // This line is the one that carries the quote-level transport into the
      // totals (resolved once, above). Every other line only stores the value.
      const isThisFreightCarrier_ = (itemIndex === freightCarrierItemIndex_);

      // ✅ NEW: Track both SENT and RECEIVED quantities (Phase 1.3)
      const existingSentQty = Number(row[colQty] || 0);
      const existingReceivedQty = actualColReceivedQty !== -1 
        ? Number(String(row[actualColReceivedQty] || '').replace(/,/g, '')) || 0 
        : 0;

      // Also track original quantities for diffing
      const existingPrice = Number(row[colPrice] || 0);
      const existingTotal = Number(row[colTotal] || 0);
      const existingStatus = String(row[colStatus] || '').trim().toUpperCase();
      const existingPn = String(row[colPn] || '').trim();
      const existingGeo = String(row[colGeo] || '').trim();
      const existingQuoteNo = colRefCotiz !== -1 ? String(row[colRefCotiz] || '').trim() : "";
      // "Ref Cotiz." is often typed as a date ("10-09-2026") which Sheets stores as a
      // real Date. Compare on a normalized key, otherwise every save reports a phantom
      // change and re-runs the stock block.
      const existingQuoteNoKey = normalizeQuoteNoForCompare_(colRefCotiz !== -1 ? row[colRefCotiz] : "");
      const existingDocUrl = colQuotedDoc !== -1 ? String(row[colQuotedDoc] || '').trim() : "";
      const existingCurrency = colCurrency !== -1 ? String(row[colCurrency] || '').trim() : "";
      const existingRate = colExchangeRate !== -1 ? (Number(row[colExchangeRate]) || 0) : 0;
      // 🔒 APPROVED QTY: "Cant." is the QUOTED quantity and freezes at Approved;
      // this cell holds what the manager actually signed off.
      const existingApprovedQty = colApprovedQty !== -1
        ? Number(String(row[colApprovedQty] || '').replace(/,/g, '')) || 0
        : 0;
      // 💰 SUPPLIER DISCOUNT + TRANSPORT (stored values). blank cell = 0, so a row
      // that predates these columns keeps its plain price and no transport.
      const existingDiscountPct = normalizeDiscountPct_(colDiscountPct !== -1 ? row[colDiscountPct] : 0);
      const existingFreight = normalizeFreight_(colFreight !== -1 ? row[colFreight] : 0);
      // Provisional freeze (final value is taken once newStatus is known below).

      // Parse receivedQty from payload (if using separate field) or from qty.
      // IMPORTANT: Only treat receivedQty as a change when the client EXPLICITLY sends
      // a value. When it is omitted (e.g. the Rcvd field is hidden for this row),
      // preserve the stored value so we never fabricate a diff or wipe existing data.
      const hasExplicitReceivedQty =
        item.receivedQty !== undefined && item.receivedQty !== null && String(item.receivedQty).trim() !== "";
      const newReceivedQty = actualColReceivedQty !== -1
        ? (hasExplicitReceivedQty ? (Number(String(item.receivedQty).replace(/,/g, '')) || 0) : existingReceivedQty)
        : 0;

      const newQty = Number(String(item.qty || item.cant || 0).replace(/,/g, '')) || 0;

      const newPrice = Number(item.price || 0);
      const newStatus = String(item.status || '').trim().toUpperCase();

      // 🔒 APPROVED QTY RESOLUTION — the heart of the quoted/approved split.
      //  * pre-approval  : the cell stays empty; the figure is stamped when the
      //                    line actually reaches Approved.
      //  * at approval   : stamp the quantity being approved (reduced or not).
      //  * post-approval : an explicit `approvedQty` from the client is a Manager
      //                    correction; otherwise the stored stamp is preserved.
      //                    A blank cell on an already-approved line means the row
      //                    predates this column, so the quoted qty is adopted.
      const hasExplicitApprovedQty =
        item.approvedQty !== undefined && item.approvedQty !== null && String(item.approvedQty).trim() !== "";
      const newApprovedOrLater = QUOTE_APPROVED_OR_LATER_.indexOf(newStatus) !== -1;
      const isApprovingNow = newStatus === 'APPROVED' && existingStatus !== 'APPROVED';
      // 🔒 THE FREEZE, evaluated on BOTH ends. Including the target status matters:
      // approving straight from this modal (Quoted -> Approved with a reduced Qty)
      // sends that reduced figure as `qty`, and without this the quoted quantity
      // would be overwritten — which is the exact data loss this split exists to
      // prevent. From here on the payload `qty` is read as the APPROVED figure.
      const quotedQtyFrozen = isQuoteQtyFrozen_(existingStatus) || newApprovedOrLater;
      let newApprovedQty = existingApprovedQty;
      if (newApprovedOrLater) {
        if (isApprovingNow || hasExplicitApprovedQty) {
          newApprovedQty = hasExplicitApprovedQty
            ? (Number(String(item.approvedQty).replace(/,/g, '')) || 0)
            : newQty;
        } else if (existingApprovedQty === 0) {
          newApprovedQty = existingSentQty; // legacy row: approved == quoted
        }
      }
      // Approving MORE than was quoted is a data-entry slip, and it would flow
      // straight into the store issue and the PO. The quoted figure is frozen at
      // this point, so it is rejected rather than silently raised.
      if (newApprovedOrLater && newApprovedQty > existingSentQty) {
        throw new Error(
          `[RECON LOCK] Approved qty (${newApprovedQty}) cannot exceed the quoted qty (${existingSentQty}).`
        );
      }

      // Which quantity the line total follows. Preserves the pre-split behaviour
      // exactly, where "Cant." doubled as both:
      //   receiving window open -> what physically arrived (hasExplicitReceivedQty)
      //   before approval        -> the quoted qty
      //   after approval         -> the APPROVED qty (previously held in "Cant.").
      // Without the last case a post-approval save — where the client echoes the
      // now-frozen quoted value back — would re-inflate the total to the quote.
      const totalBasisQty = hasExplicitReceivedQty
        ? newReceivedQty
        : (newApprovedOrLater ? newApprovedQty : newQty);

      // 💰 DISCOUNT + TRANSPORT RESOLUTION.
      // `undefined` = the client did not touch the field, so the stored value is
      // preserved; an explicit value is validated and applied. Unlike the quoted
      // quantity these NEVER freeze (a supplier can revise its discount, or add a
      // freight charge, at any point before the goods move), but the USER role is
      // read-only on both - same protection the price already has.
      const hasExplicitDiscountPct =
        item.discountPct !== undefined && item.discountPct !== null && String(item.discountPct).trim() !== "";
      let newDiscountPct = existingDiscountPct;
      if (hasExplicitDiscountPct) {
        const rawPct = Number(String(item.discountPct).replace(/[%\s,]/g, "").replace(",", "."));
        if (!isFinite(rawPct) || isNaN(rawPct)) {
          throw new Error(`[RECON LOCK] Discount for GEO ${item.origGeo || item.geo || "?"} is not a number: "${item.discountPct}".`);
        }
        if (rawPct < 0 || rawPct > QUOTE_MAX_DISCOUNT_PCT_) {
          throw new Error(`[RECON LOCK] Discount for GEO ${item.origGeo || item.geo || "?"} must be between 0% and 100% (got ${rawPct}%).`);
        }
        newDiscountPct = normalizeDiscountPct_(rawPct);
      }

      const hasExplicitFreight =
        item.freight !== undefined && item.freight !== null && String(item.freight).trim() !== "";
      let newFreight = existingFreight;
      if (hasExplicitFreight) {
        const rawFreight = Number(String(item.freight).replace(/[\s,]/g, "").replace(",", "."));
        if (!isFinite(rawFreight) || isNaN(rawFreight)) {
          throw new Error(`[RECON LOCK] Transport for GEO ${item.origGeo || item.geo || "?"} is not a number: "${item.freight}".`);
        }
        if (rawFreight < 0) {
          throw new Error(`[RECON LOCK] Transport for GEO ${item.origGeo || item.geo || "?"} cannot be negative (got ${rawFreight}).`);
        }
        if (rawFreight > QUOTE_MAX_FREIGHT_) {
          throw new Error(`[RECON LOCK] Transport for GEO ${item.origGeo || item.geo || "?"} exceeds the maximum allowed (${QUOTE_MAX_FREIGHT_}).`);
        }
        newFreight = normalizeFreight_(rawFreight);
      }

      // 🛎️ FIX (approval totals): the sheet owns "Total Neto" arithmetic. Clients
      // used to forward whatever total they had cached, so approving a quote with
      // an edited Qty wrote the new Cant. while keeping the OLD Total - the line
      // total and the quote's grand total therefore never updated.
      //   Total Neto = (Precio Unit. - Descuento %) x <basis qty>  [+ Transporte]
      // A zero unit price means the supplier quoted a lump sum, so whatever the
      // payload carries is preserved untouched rather than being zeroed out.
      //
      // The net unit price is derived (the quoted price is never overwritten), so
      // the discount automatically follows every later edit of the price. Freight
      // is a quote-level charge stored on every row but added to the totals ONCE,
      // by the carrier row only (see QUOTE_FREIGHT_HEADER_).
      const newNetUnitPrice = resolveNetUnitPrice_(newPrice, newDiscountPct);
      const newTotal = newPrice > 0
        ? resolveQuoteLineTotal_(newNetUnitPrice, totalBasisQty, newFreight, isThisFreightCarrier_)
        : (Number(item.total || 0) || 0);
      const newPn = String(item.pn || '').trim();
      const newGeo = String(item.geo || '').trim();
      const newSupplierQuoteNo = String(item.supplierQuoteNo || existingQuoteNo).trim();
      // `quotedDocUrl` is per line: each item keeps its own supplier quotation file,
      // so a supplier quoting part of the items in one file and the rest in another
      // ends up with a different document per row. An omitted/empty value PRESERVES
      // the stored link; only the explicit `clearQuotedDoc` flag blanks the cell
      // (that is what the per-line ✕ in the edit modal sends).
      const newQuotedDocUrl = item.clearQuotedDoc === true
        ? ''
        : String(item.quotedDocUrl || existingDocUrl).trim();

      // 🔒 MANDATORY PROOF FOR "Quoted": a line saved as Quoted MUST carry BOTH
      // the Supplier Quote No. and the supplier quotation PDF (the PDF is only
      // ever uploaded at the Quoted moment). Covers every entry path (edit
      // modal, compare-apply, resync, direct calls); the client blocks first,
      // this is the bypass-proof backstop.
      {
        const toStatus = String(newStatus || '').trim().toUpperCase();
        if (toStatus === 'QUOTED' && String(item.deleted || '') !== 'true' && item.deleted !== true) {
          const hasNo = String(newSupplierQuoteNo || '').trim() !== '';
          const hasDoc = String(newQuotedDocUrl || '').trim() !== '' && /^https?:\/\//i.test(String(newQuotedDocUrl || '').trim());
          if (!hasNo || !hasDoc) {
            throw new Error('[RECON LOCK] GEO ' + (newGeo || item.origGeo || '?') + ' saved as Quoted requires Supplier Quote No. AND the quotation document (' + (!hasNo ? 'missing Quote No.' : '') + (!hasNo && !hasDoc ? ' + ' : '') + (!hasDoc ? 'missing document' : '') + ').');
          }
        }
      }

      // Currency / exchange rate (metadata only): preserve the stored value unless the
      // payload carries an explicit new one. Price/Total stay in the supplier currency.
      let newCurrency = existingCurrency;
      if (item.currency !== undefined && item.currency !== null && String(item.currency).trim() !== '') {
        newCurrency = String(item.currency).trim();
      }
      let newRate = existingRate;
      const hasExplicitRate = item.exchangeRate !== undefined && item.exchangeRate !== null && String(item.exchangeRate).trim() !== '';
      if (hasExplicitRate) {
        const parsedRate = Number(item.exchangeRate);
        if (isFinite(parsedRate) && parsedRate >= 0) newRate = parsedRate;
      }

      const itemDiff = { index: rowIndex, quoteId: item.quoteId, origGeo: item.origGeo, diffs: {} };

      // ✅ NEW: Track diffs for both SENT and RECEIVED quantities (Phase 1.4)
      
      // Check sent quantity changes (should only happen by warehouse/manager)
      // 🔒 Never reported for a frozen line. The client echoes the stored quoted
      // qty back, and with the receiving window open it sends the RECEIVED quantity
      // as `qty` (QuoteMgmtJS.html) — neither is a re-quote, and reporting them
      // would trigger a spurious PDF regeneration and stock run on every save.
      if (!quotedQtyFrozen && Math.abs(existingSentQty - newQty) > 1e-6) {
        itemDiff.diffs.sentQty = {
          from: existingSentQty,
          to: newQty,
          reason: 'sent_qty_changed'
        };
        hasQtyAmountChange = true;
        hasAnyChange = true;
      }

      // Check approved quantity changes (stamped at approval, corrected after).
      // On a legacy row the first save backfills the column, which is why this
      // also raises hasAnyChange: without it the early "no changes" return would
      // skip the batch write and the backfill would never be persisted.
      if (colApprovedQty !== -1 && Math.abs(existingApprovedQty - newApprovedQty) > 1e-6) {
        itemDiff.diffs.approvedQty = {
          from: existingApprovedQty,
          to: newApprovedQty,
          reason: isApprovingNow ? 'approved_qty_stamped' : 'approved_qty_changed'
        };
        hasQtyAmountChange = true;
        hasAnyChange = true;
      }

      // Check received quantity changes (allowed for USER role after Despatched)
      if (actualColReceivedQty !== -1 && Math.abs(existingReceivedQty - newReceivedQty) > 1e-6) {
        itemDiff.diffs.receivedQty = { 
          from: existingReceivedQty, 
          to: newReceivedQty,
          reason: 'received_qty_changed'
        };
        hasQtyAmountChange = true;
        hasAnyChange = true;
      }

      // Check price changes (restricted for USER before Despatched)
      if (Math.abs(existingPrice - newPrice) > 1e-6) {
        itemDiff.diffs.price = { 
          from: existingPrice, 
          to: newPrice,
          reason: 'price_changed'
        };
        hasQtyAmountChange = true;
        hasAnyChange = true;
      }

      // Check total changes (auto-calculated based on received qty)
      if (Math.abs(existingTotal - newTotal) > 1e-6) {
        itemDiff.diffs.total = { 
          from: existingTotal, 
          to: newTotal,
          reason: 'total_changed'
        };
        hasQtyAmountChange = true;
        hasAnyChange = true;
      }

      // 💰 Discount / transport changes. Both move money, so both count as an
      // amount change: without this, entering a discount or a freight charge
      // would leave the generated quote PDF showing the stale (undiscounted)
      // subtotal, because only hasQtyAmountChange triggers a regeneration.
      if (Math.abs(existingDiscountPct - newDiscountPct) > 1e-9) {
        itemDiff.diffs.discountPct = {
          from: existingDiscountPct,
          to: newDiscountPct,
          reason: 'discount_changed'
        };
        hasQtyAmountChange = true;
        hasAnyChange = true;
      }
      if (Math.abs(existingFreight - newFreight) > 1e-9) {
        itemDiff.diffs.freight = {
          from: existingFreight,
          to: newFreight,
          reason: 'freight_changed'
        };
        hasQtyAmountChange = true;
        hasAnyChange = true;
      }

      if (existingStatus !== newStatus) {
        // 🔒 ADDED: Strict Sequential Status Forward-Only Enforcement
        if (!isValidQuoteItemStatusProgression(existingStatus, newStatus)) {
          throw new Error(`Invalid status transition from "${existingStatus}" to "${newStatus}". Status must move forward sequentially.`);
        }

        // 🔒 ISSUE-4: cross-supplier double-approval guard (server-side twin of
        // the compare modal's approvedByItem lock). Client state can be stale
        // or hand-built (e.g. the Edit modal's bulk "Apply to selected"), so
        // moving a line INTO Approved must fail when a DIFFERENT supplier
        // already holds this exact item (same Ref ID + GEO/PN) at
        // Approved-or-later — exactly the gap the UI block used to leave once
        // the winner advanced to Sent/Received/.... Same-supplier rows (e.g. a
        // re-sent RFQ round) and Deleted rows never block.
        if (newStatus === 'APPROVED' && existingStatus !== 'APPROVED') {
          // Caller context for the rival check (see findApprovedRival_).
          // Supplier: the row being saved carries the supplier value in the
          // payload item first, else the quote-level payload supplier, else the
          // stored row value. GEO/PN likewise prefer the stored row cells so a
          // mid-edit rename cannot dodge the match; blank-blank = unchecked.
          const rivalCtxSupplier = String(item.supplier || payload.supplier || (typeof row !== 'undefined' && colSupplier !== -1 ? row[colSupplier] : '') || '').trim();
          const rivalCtxGeo = String(existingGeo || item.geo || '').trim();
          const rivalCtxPn = String(existingPn || item.pn || item.p || '').trim();
          const rivalCtxOrder = String(item.orderId || payload.orderId || '').trim();
          // Same double guard against sheet state: read the stored Ref ID cell
          // (a payload with a blank orderId must not match half the sheet).
          const storedOrderForGuard = (typeof row !== 'undefined' && colRefId !== -1) ? String(row[colRefId] || '').trim() : '';
          const doubleWin = (rivalCtxGeo || rivalCtxPn) && (rivalCtxOrder || storedOrderForGuard) ? findApprovedRival_(rows || [], rowIndex, {
            col: function () {
              // Reuse updateQuoteWithItems' own header map (headers/findCol are
              // in scope here): Ref ID for the requirement, Proveedor for the
              // supplier, Geo Code + P/N for the item key, Estado for status.
              const names = Array.prototype.slice.call(arguments);
              for (let n = 0; n < names.length; n++) {
                const want = String(names[n] || '').trim().toLowerCase();
                const hit = headers.findIndex(function (h) { return String(h || '').trim().toLowerCase() === want; });
                if (hit !== -1) return hit;
              }
              return -1;
            },
            orderId: rivalCtxOrder || storedOrderForGuard,
            supplier: rivalCtxSupplier,
            geo: rivalCtxGeo,
            pn: rivalCtxPn,
          }) : null;
          if (doubleWin) {
            throw new Error(
              `[RECON LOCK] Item ${existingGeo || existingPn || '?'} of requirement ${payload.orderId || ''} is already approved from "${doubleWin.supplier}" (status ${doubleWin.status}). ` +
              `Mark that quote line as Deleted first if the award really moved to another supplier.`
            );
          }
        }

/**
 * Get email address for a user by looking up in the Users sheet cache
 * Searches by username/name in column A or B, returns email from column D
 * Returns empty string if not found
 */
function getUserEmailFromCache(userName) {
  console.log("DEBUG getUserEmailFromCache: ENTER function with userName:", userName);
  try {
    if (!userName || userName.trim() === "") {
      console.log("DEBUG getUserEmailFromCache: userName is empty, returning empty string");
      return "";
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const usersSheet = ss.getSheetByName("Users");
    if (!usersSheet) {
      console.warn("Users sheet not found");
      return "";
    }

    const data = usersSheet.getDataRange().getValues();
    const normalizedSearch = userName.toLowerCase().trim();
    console.log("DEBUG getUserEmailFromCache: Looking for user:", userName, "->", normalizedSearch);

    // Find user by name in column C (Full Name), return email from column D
    for (let i = 1; i < data.length; i++) {
      const fullName = String(data[i][2] || "").toLowerCase().trim();  // Column C (index 2)

      if (i < 5) {
        console.log(`DEBUG getUserEmailFromCache: Row ${i} - Full Name: "${fullName}", Email: "${data[i][3]}"`);
      }

      if (fullName === normalizedSearch) {
        // Return email from column D (index 3)
        const email = String(data[i][3] || "").trim();
        console.log("DEBUG getUserEmailFromCache: Found user! Email:", email);
        return email;
      }
    }

    console.warn("User not found in Users sheet:", userName, "searched:", normalizedSearch);
    return "";
  } catch (err) {
    console.error("Error in getUserEmailFromCache:", err.toString());
    return "";
  }
}

// Tokens that mean "no real category has been assigned". Mirrors
// window.GENERAL_TOKENS in CreateReqJS.html / RFQJS.html so the server and the
// client agree on what counts as uncategorized.
const GENERAL_CATEGORY_TOKENS_ = [
  "", "general", "generica", "generico", "sin categoria",
  "sin cat", "uncategorized", "varios", "n/a", "na", "none", "-",
];

function isGeneralCategoryValue_(cat) {
  return GENERAL_CATEGORY_TOKENS_.indexOf(String(cat || "").trim().toLowerCase()) !== -1;
}

/**
 * Resolves the category shown for a requirement line on the RFQ page.
 *
 * The Ordenes "Category" column is authoritative: processMassOrders writes
 * shipment.category there (CreateReqGS.js) and it is the value the
 * category-assignment dialog collected at requirement creation - it is also
 * embedded in the Req ID (REQ-0750-GEN-MANTENIMIENTO).
 *
 * Master column C is only a FALLBACK, for legacy rows whose Ordenes category is
 * blank. Previously this function read Master exclusively and defaulted to the
 * literal "Uncategorized", so any GEO with a blank/missing Master category was
 * split out into a second RFQ category card and a second Quotes group - even
 * though Ordenes already carried the correct category. That produced one
 * requirement appearing as a "16-item" quote plus an "Uncategorized" remainder.
 *
 * @param {string} ordenesCategory category from the Ordenes row (item.category)
 * @param {string} masterCategory  category from the Master sheet (official.category)
 * @return {string} never blank, never "Uncategorized"
 */
function resolveRequirementCategory_(ordenesCategory, masterCategory) {
  const fromOrdenes = String(ordenesCategory || "").trim();
  if (!isGeneralCategoryValue_(fromOrdenes)) return fromOrdenes;
  const fromMaster = String(masterCategory || "").trim();
  if (!isGeneralCategoryValue_(fromMaster)) return fromMaster;
  return "General";
}

function getPendingRequirementsForRFQ(site, statusFilter) {
  const allHistory = getRequirementsHistory();
  const masterData = getInventoryData();

  const masterMap = new Map();
  masterData.forEach((item) => {
    // We add 'supplier' to the map retrieval
    masterMap.set(item.geo, {
      en: item.en,
      es: item.es,
      supplier: item.proveedor,
      category: item.category,
      subCategory: item.sub_categoria,
    });
  });

  // statusFilter: "Approved" (default) | "All" (everything except Pending, i.e.
  // requirements still awaiting approval are excluded) | any specific status
  // word (Quoted, Sent, Received, Despatched, ...).
  const statusKey = String(statusFilter || "Approved").trim().toLowerCase();
  let pending;
  if (statusKey === "all") {
    pending = allHistory.filter(
      (item) =>
        item.estado &&
        !item.estado.toString().toLowerCase().includes("pend"),
    );
  } else {
    pending = allHistory.filter(
      (item) =>
        item.estado && item.estado.toString().toLowerCase().includes(statusKey),
    );
  }
  if (site) {
    pending = pending.filter((item) => item.faena === site);
  }

  // Existing quotes per requirement item (for the "additional quote" flow):
  // lets the UI show which suppliers already quoted each GEO and block
  // re-selecting the same supplier for the same item.
  const quotesByOrderGeo = {};
  try {
    (getQuotesHistory() || []).forEach(function (q) {
      if (String(q.estado || "").toLowerCase() === "deleted") return;
      const key = String(q.orderId || "").trim() + "||" + String(q.geo || "").trim();
      (quotesByOrderGeo[key] = quotesByOrderGeo[key] || []).push({
        supplier: q.supplier || "",
        quoteId: q.quoteId || "",
        status: q.estado || "",
      });
    });
  } catch (e) {
    console.warn("RFQ quote-info join failed: " + e);
  }

  const groupedReqs = {};
  pending.forEach((item) => {
    if (!groupedReqs[item.order]) {
      groupedReqs[item.order] = {
        id: item.order,
        date: item.fecha,
        site: item.faena,
        subArea: item.subArea || "",
        status: item.estado || "",
        items: [],
      };
    }

    const official = masterMap.get(item.geo) || {
      en: "",
      es: "",
      supplier: "",
      category: "",
      subCategory: "",
    };
    groupedReqs[item.order].items.push({
      p: item.pn || "",
      geo: item.geo || "",
      // Capture both descriptions clearly
      es: (official.es || "").toString().trim(),
      en: (official.en || "").toString().trim(),
      // Keep the raw requirement description as a safety reference
      req_desc: (item.desc || "").toString().trim(),
      qty: item.cant || 0,
      supplier: official.supplier || "",
      category: resolveRequirementCategory_(item.category, official.category),
      subCategory: official.subCategory || "",
      quotedBy: quotesByOrderGeo[
        String(item.order || "").trim() + "||" + String(item.geo || "").trim()
      ] || [],
      // Item notes are unique to each requirement, so the Req ID travels with
      // the note: consolidated rows can render "Req 100123: <note>".
      reqItemNotes: [
        {
          ref: String(item.order || "").trim(),
          notes: String(item.itemNotes || ""),
        },
      ],
      // Supplier-facing notes travel the same way. These are the ONLY per-item
      // notes printed in the supplier RFQ email ("Item Notes" stays internal).
      reqSupplierNotes: [
        {
          ref: String(item.order || "").trim(),
          notes: String(item.supplierNotes || ""),
        },
      ],
      // Per-item photos (Ordenes "Item Images": comma-separated Drive URLs).
      // Kept per source Req (raw cell string) so the RFQ table, the preview and
      // the supplier email can render the requester's attachments. The client
      // unions them when several requirements are consolidated onto one GEO.
      images: String(item.images || ""),
    });
  }

function updateUserPin(userIdentifier, oldPin, newPin) {

  try {

    if (!userIdentifier || !oldPin || !newPin) {
      return {
        success: false,
        message: "Missing required fields."
      };
    }

    oldPin = String(oldPin).trim();
    newPin = String(newPin).trim();
    userIdentifier = String(userIdentifier).trim().toLowerCase();

    if (!/^\d{4}$/.test(oldPin)) {
      return {
        success: false,
        message: "Current PIN must contain exactly 4 digits."
      };
    }

    if (!/^\d{4}$/.test(newPin)) {
      return {
        success: false,
        message: "New PIN must contain exactly 4 digits."
      };
    }

    if (oldPin === newPin) {
      return {
        success: false,
        message: "New PIN must be different from your current PIN."
      };
    }

    const sheet = SpreadsheetApp
      .getActiveSpreadsheet()
      .getSheetByName("Users");

    if (!sheet) {
      return {
        success: false,
        message: "Users sheet not found."
      };
    }

    const data = sheet.getDataRange().getValues();

    let row = -1;

    for (let i = 1; i < data.length; i++) {

      const userId = String(data[i][0]).trim().toLowerCase();
      const email = String(data[i][3]).trim().toLowerCase();

      if (userId === userIdentifier || email === userIdentifier) {
        row = i;
        break;
      }

    }

    if (row === -1) {
      return {
        success: false,
        message: "User not found."
      };
    }

    const currentPin = String(data[row][1]).trim();

    if (currentPin !== oldPin) {
      return {
        success: false,
        message: "Current PIN is incorrect."
      };
    }

    sheet.getRange(row + 1, 2).setValue(newPin);

    return {
      success: true,
      message: "PIN updated successfully."
    };

  } catch (err) {

    return {
      success: false,
      message: err.message
    };

  }

    sheet
      .getRange(rowIndex, 1, 1, 3)

function uploadQuoteAttachment(payload) {
  try {
    if (!payload || !Array.isArray(payload.files) || payload.files.length === 0) {
      return { success: false, error: "No file content provided." };
    }

    const folder = getQuoteDocsFolder_(payload);
    const baseName = sanitizeFileName_(
      String(payload.fileName || "QuotedDoc")
        .replace(/\.[^.]+$/, "")
        .trim() || "QuotedDoc"
    );

    // Single file (image or PDF) → store directly as-is.
    if (payload.files.length === 1) {
      const f = payload.files[0];
      const mime = String(f.mimeType || "").toLowerCase();
      const ext = mime.indexOf("pdf") !== -1 ? ".pdf"
        : mime.indexOf("image/") !== -1 ? ".jpg"
        : String(payload.fileName || "").replace(/.*(\.[^.]+)$/, "$1") || ".bin";
      const blob = Utilities.newBlob(
        Utilities.base64Decode(f.base64),
        String(f.mimeType || ""),
        baseName + ext
      );
      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      return { success: true, url: file.getUrl(), fileName: file.getName() };
    }

    // Fallback: multiple pages/images → repackage into one compact PDF via the
    // Drive HTML→PDF engine (not normally hit; client sends single files only).
    const imgTags = payload.files.map(function (f) {
      return '<img src="data:' + String(f.mimeType || "image/jpeg") + ';base64,' +
        String(f.base64 || "") + '" ' +
        'style="width:100%; display:block; margin:0; page-break-after:always;">';
    }).join("\n");

    const html =
      '<!DOCTYPE html><html><head><meta charset="utf-8">' +
      '<style>body{margin:0;padding:0;}img{max-width:100%;}</style></head>' +
      '<body>' + imgTags + '</body></html>';

    const htmlBlob = Utilities.newBlob(html, MimeType.HTML, baseName + ".html");
    const htmlFile = folder.createFile(htmlBlob);
    const pdfBlob = htmlFile.getAs(MimeType.PDF);

    if (!pdfBlob || pdfBlob.getBytes().length === 0) {
      throw new Error("PDF conversion failed from document pages");
    }

    const pdfFile = folder.createFile(pdfBlob);
    htmlFile.setTrashed(true);
    pdfFile.setName(baseName + ".pdf");
    pdfFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    return { success: true, url: pdfFile.getUrl(), fileName: pdfFile.getName() };
  } catch (e) {
    console.warn("uploadQuoteAttachment failed: " + e.toString());
    return { success: false, error: String(e) };
  }
}
